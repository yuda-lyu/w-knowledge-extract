const PREFIX_OF_KEY = { claims: 'C', disputes: 'D', parameters: 'P', questions: 'Q' }

/**
 * 單一操作之長度問題(與 applyDelta 同一判準、同一欄位表:overLength／essenceOver 為單一來源)——供工作流於套用前預判:
 * 超限者轉為程式意見交修訂縮寫、修訂補丁不得使本條由合格變超限、定稿前本質仍超限者另行處置(extract/workflow/runPass)。
 * 只涵蓋 essence、add、revise、param_add、question_add、dispute_add／dispute_update;其餘操作回空字串(仍以 applyDelta 為準)。
 *
 * @param {Object} op 輸入差量操作
 * @param {Object} [limits] 輸入長度上限，逐鍵覆寫 DEFAULT_LIMITS
 * @returns {String} 回傳第一個長度問題(與 applyDelta 拒收理由同字樣)，無問題回空字串

    const lim = { ...DEFAULT_LIMITS, ...(isobj(limits) ? limits : {}) }
    const over = (key, obj, fields) => {
        for (const f of fields || TEXT_FIELDS[key] || []) {
            for (const v of (ARRAY_FIELDS.has(f) ? strs(obj?.[f]) : [str(obj?.[f])])) {
                const p = overLength(f, v, lim)
                if (p) return p
            }
        }
        return ''
    }
    switch (str(op?.op)) {
        case 'essence': return essenceOver(str(op.text), lim)
        case 'add': return over('claims', op)
        case 'revise': {
            const key = KEY_OF_PREFIX[str(op.id).charAt(0)] || 'claims'
            const f = isobj(op.fields) ? op.fields : {}
            return over(key, f, Object.keys(f).filter((k) => (TEXT_FIELDS[key] || []).includes(k)))
        }
        case 'param_add': return over('parameters', op)
        case 'question_add': return over('questions', { text: op.text })
        case 'dispute_add':
        case 'dispute_update': {
            for (const sd of isarr(op.sides) ? op.sides : []) {
                const p = over('disputes', sd, ['position', 'conditions', 'basis'])
                if (p) return p
            }
            return over('disputes', op.op === 'dispute_add' ? { question: op.question } : { note: op.note })
        }
        default: return ''
    }
}

/**
 * 套用差量:逐操作原子(不合規者拒收並記原因,其餘照套);新項 id 由程式配發(單調、永不重用);出處只增;
 * 同一差量內之新項可以 ref 互引(他處寫 '@ref');規則見 stores/coreState 檔頭【操作語意】
 *
 * @param {Object} state 輸入目前狀態(不修改,回傳新狀態)
 * @param {Object} delta 輸入差量 { ops:[...] }(skipped 等其他欄位由呼叫端處理)
 * @param {Object} [ctx={}] 輸入脈絡，非物件視為{}
 * @param {Array} [ctx.batch=[]] 輸入本批筆記 [{ code:'N1', id }]；出處只能引本批(以代號或 id 皆可)
 * @param {String} [ctx.mode='delta'] 輸入 'delta'｜'consolidate'
 * @param {Array} [ctx.onlyOps] 輸入本 pass 只收之操作名(為該模式可用操作之子集;如本質專步 ['essence']),其餘拒收
 * @param {Array} [ctx.claimKinds] 輸入許可之主張種類，預設 DEFAULT_CLAIM_KINDS
 * @param {Object} [ctx.limits] 輸入長度上限，逐鍵覆寫 DEFAULT_LIMITS
 * @param {Array} [ctx.withhold=[]] 輸入本次暫緩之操作名(如終審失敗之降級:終態操作不套用)，被暫緩者記入 withheld(帶 resolved＝@ref 已換實際 id 之操作)；
 *   個別操作帶 _withhold:true 者亦暫緩(終審漏裁決之終態操作)
 * @param {Function} [ctx.keyOf] 輸入概念名正規化函數(related 去重用)，預設去空白小寫
 * @param {String} [ctx.self=''] 輸入本核心之概念名(related 去自身用)
 * @param {String} [ctx.at=''] 輸入時刻字串(寫入 history 與 changelog)
 * @param {Object} [ctx.meta={}] 輸入寫入 changelog 之附加欄位(如 provider、reviewed)
 * @param {Array} [ctx.recheckIds=[]] 輸入本批中之重審筆記 id(內容已更新或新衝突之夥伴):重審批內,尚有重審筆記以外之存活出處者
 *   不得以「無出處支持」撤回(改寫只可 revise——起草者看不到他篇出處,撤回不可逆)
 * @param {Boolean} [ctx.program=false] 輸入程式差量(⑱;檢定標準 ES9 存量、ES8):須給 onlyOps;可用全部操作、revise 不需本批出處、
 *   撤回可用 PROGRAM_RETRACT_REASONS
 * @returns {Object} 回傳 { state, applied:Array({index,op,id?,sources,noop?}), rejected:Array({index,op,reason,sources}), withheld:Array({index,op,sources}), touched:Array(id), overLimit:Integer(逾建議上限而放行之欄數), lengthChecked:Integer(受長度檢查之欄數) }
 * @example
 * const ops = [{ op: 'add', ref: 'r0', kind: 'rule', text: '甲', basis: '案例', sources: ['N1'], critique: '僅單一案例', pros: ['出處未載'], cons: ['出處未載'] }]
 * const r = applyDelta(emptyState(), { ops }, { batch: [{ code: 'N1', id: 'n-a' }] })
 * console.log(r.applied.length, r.rejected.length, r.state.claims.map((c) => [c.id, c.text, c.sources]))
 * // => 1 0 [ [ 'C1', '甲', [ 'n-a' ] ] ]
 */
export function applyDelta(state, delta, ctx = {}) {
    if (!isobj(ctx)) ctx = {}
    const s = clone(state)
    const recheckIds = new Set(isarr(ctx.recheckIds) ? ctx.recheckIds.map((x) => str(x)) : [])
    const W = withdrawnOf(s)
    const unsup0 = unsupportedOf(s)
    const mode = ctx.mode === 'consolidate' ? 'consolidate' : 'delta'
    // ctx.onlyOps:本 pass 只收之操作(如本質專步只收 essence);其餘一律拒收並寫明——不在本 pass 之職責內者不夾帶入核心
    const only = isarr(ctx.onlyOps) && ctx.onlyOps.length ? ctx.onlyOps : null
    // 程式差量(⑱)須限定操作(onlyOps),否則不啟用
    const program = ctx.program === true && !!only
    const allowed = (program ? [...ALL_OPS] : (mode === 'consolidate' ? CONSOLIDATE_OPS : DELTA_OPS)).filter((x) => !only || only.includes(x))
    const retractReasons = program ? [...RETRACT_REASONS, ...PROGRAM_RETRACT_REASONS] : RETRACT_REASONS
    const kinds = isarr(ctx.claimKinds) && ctx.claimKinds.length ? ctx.claimKinds : DEFAULT_CLAIM_KINDS
    const limits = { ...DEFAULT_LIMITS, ...(isobj(ctx.limits) ? ctx.limits : {}) }
    const withhold = new Set(isarr(ctx.withhold) ? ctx.withhold : [])
    const keyOf = typeof ctx.keyOf === 'function' ? ctx.keyOf : (x) => String(x || '').replace(/\s+/g, '').toLowerCase()
    const at = str(ctx.at)
    const version = s.version + 1
    const ops = isobj(delta) && isarr(delta.ops) ? delta.ops : []

    // 本批代號 → 筆記 id(亦接受完整 id)
    const codeMap = codeMapOf(ctx.batch)
    const toNote = (x) => noteOfCode(codeMap, x)
    const srcOf = (v) => uniq(strs(v).map(toNote).filter(Boolean))
    const preIds = new Set(allItemsOf(s).map((x) => x.id))

    const applied = []
    const rejected = []
    const withheld = []
    const touched = new Set()
    let overLimit = 0
    let lengthChecked = 0
    const opSources = (op) => uniq([...srcOf(op?.sources), ...(isarr(op?.sides) ? op.sides.flatMap((sd) => srcOf(sd?.sources)) : []), ...(isarr(op?.sides_add) ? op.sides_add.flatMap((sd) => srcOf(sd?.sources)) : [])])
    // 回報用序號:經 applyVerdicts 剔除後之操作帶 _i(差量原序號),回報一律用原序號
    const rep = (index, op) => (Number.isInteger(op?._i) ? op._i : index)
    // kind＝拒收之型別碼(core/rejectKinds 之 opsRejected 組;每個分支之穩定識別,未給或未登記者歸 unclassified);reason 之字樣不變
    const reject = (index, op, reason, kind) => {
        rejected.push({ index: rep(index, op), op: str(op?.op) || '?', reason, sources: opSources(op), kind: rejectKindOf('opsRejected', kind) })
    }

    // ── 前置掃描:ref 登記(重複者皆拒)、跨操作衝突(既有項至多一個改狀態操作;confirm 與終態操作不並存)──
    const refState = new Map() // ref → { id:null|String, rejected:Boolean }
    const refDup = new Set()
    ops.forEach((op) => {
        const ref = str(op?.ref)
        if (!ref || !['add', 'dispute_add', 'param_add', 'question_add'].includes(op?.op)) return
        if (refState.has(ref)) refDup.add(ref)
        else refState.set(ref, { id: null, rejected: false })
    })
    // 被終審剔除之新項(applyVerdicts 之 droppedRefs):引用它們之保留操作拒收時寫明「已被終審剔除(理由)」,
    //   不以「不存在」之類之理由掩蓋上游之剔除;同名 ref 仍由存活操作定義者不登記
    for (const [ref, why] of Object.entries(isobj(ctx.droppedRefs) ? ctx.droppedRefs : {})) {
        if (!refState.has(ref)) refState.set(ref, { id: null, rejected: true, dropped: str(why) || '終審剔除' })
    }
    const stateOps = new Map() // 既有 id → 改狀態操作之索引
    const confirmOps = new Map()
    const terminalOps = new Map()
    ops.forEach((op, index) => {
        if (!isobj(op)) return
        const targets = op.op === 'merge' ? strs(op.from) : (['revise', 'supersede', 'retract', 'contest', 'dispute_dissolve', 'split'].includes(op.op) ? [str(op.id)] : [])
        for (const t of targets) {
            if (!preIds.has(t)) continue
            stateOps.set(t, [...(stateOps.get(t) || []), index])
            if (TERMINAL_OPS.includes(op.op)) terminalOps.set(t, [...(terminalOps.get(t) || []), index])
        }
        if (op.op === 'confirm' && preIds.has(str(op.id))) confirmOps.set(str(op.id), [...(confirmOps.get(str(op.id)) || []), index])
    })
    const conflict = new Map() // index → 原因
    for (const [t, idx] of stateOps) if (idx.length > 1) for (const i of idx) conflict.set(i, { why: `同一項目 ${t} 有多個改狀態操作`, kind: 'state-op-multiple' })
    for (const [t, idx] of terminalOps) {
        if (!confirmOps.has(t)) continue
        for (const i of [...idx, ...confirmOps.get(t)]) conflict.set(i, { why: `同一項目 ${t} 同時被確認與取代／撤回`, kind: 'state-op-confirm-vs-terminal' })
    }
    // 失撐條目之補出處:同一差量內 confirm 帶本批出處者視為已補,洗白守門對其放行
    const resourced = new Set()
    if (unsup0.any) ops.forEach((op) => { if (isobj(op) && op.op === 'confirm' && unsup0.items.has(str(op.id)) && srcOf(op.sources).length) resourced.add(str(op.id)) })
    // 洗白守門:出處已撤之條目不得被合併(from／into)、不得被新本質、新爭議方或 contest 引用——
    //   否則失撐之內容經合併或引用,以他條之出處呈現;同一差量內已 confirm 補出處者除外
    const launderOf = (op) => {
        if (!unsup0.any) return ''
        let ids = []
        if (op.op === 'merge') ids = [...strs(op.from), str(op.into)]
        else if (op.op === 'essence') ids = strs(op.claims)
        else if (op.op === 'contest') ids = [str(op.id)]
        else if (op.op === 'dispute_add' || op.op === 'dispute_update') ids = [...(isarr(op.sides) ? op.sides : []), ...(isarr(op.sides_add) ? op.sides_add : [])].flatMap((sd) => strs(sd?.claims))
        const hit = uniq(ids.filter((x) => x && !x.startsWith('@') && unsup0.items.has(x) && !resourced.has(x)))
        const heldHit = hit.filter((x) => unsup0.held.has(x))
        const lostHit = hit.filter((x) => !unsup0.held.has(x))
        const why = [
            lostHit.length ? `所引〔${lostHit.join('〕〔')}〕之出處已撤（筆記已刪除），須先以本批筆記 confirm 補出處` : '',
            heldHit.length ? `所引〔${heldHit.join('〕〔')}〕出處支持未能判定、已隔離待判，不得引用` : '',
        ].filter(Boolean).join('；')
        // 兩種並見者以先列者(出處已撤)為碼
        return why ? { why, kind: lostHit.length ? 'launder-source-withdrawn' : 'launder-source-held' } : ''
    }
    // 重審守門:重審批內,尚有重審筆記以外之存活出處者不得以「無出處支持」撤回(起草者看不到他篇出處;撤回不可逆)
    const recheckRetractOf = (op) => {
        if (op.op !== 'retract' || !recheckIds.size || str(op.reason) !== '無出處支持') return ''
        const h = findItem(s, str(op.id))
        if (!h) return ''
        const others = liveSourcesOf(h.item, s, W).filter((x) => !recheckIds.has(x))
        return others.length ? { why: `〔${h.item.id}〕仍有他篇出處（${others.length} 篇），不可以「無出處支持」撤回——只可 revise`, kind: 'recheck-retract-has-other-sources' } : ''
    }

    // 處理順序:新增主張 → 新增爭議／參數／問題 → 其餘依原序(「先引用、後定義」亦可解析)
    const phaseOf = (op) => (op?.op === 'add' ? 0 : ['dispute_add', 'param_add', 'question_add'].includes(op?.op) ? 1 : 2)
    const order = ops.map((op, index) => ({ op, index })).sort((a, b) => phaseOf(a.op) - phaseOf(b.op) || a.index - b.index)
    const newId = (key) => {
        const p = PREFIX_OF_KEY[key]
        const n = Math.max(Number(s.nextId[p]) || 1, 1)
        s.nextId[p] = n + 1
        return `${p}${n}`
    }
    // 解析引用:'@ref' → 已配發之 id;所依之新項未成立 → 依賴失敗,dep 為寫明根因之拒收理由(被終審剔除／被程式拒收);一般 id 須存在
    const resolveRef = (x) => {
        const v = str(x)
        if (v.startsWith('@')) {
            const r = refState.get(v.slice(1))
            if (!r) return { id: null, reason: `ref「${v}」未定義`, kind: 'ref-undefined' }
            if (!r.id) return { id: null, reason: 'dependency-rejected', kind: 'dependency-rejected', dep: r.dropped ? `所引 ${v} 已被終審剔除（${r.dropped}）` : `所引 ${v} 已被拒收（${r.why || '未成立'}）` }
            return { id: r.id }
        }
        return findItem(s, v) ? { id: v } : { id: null, reason: `引用之項目 ${v || '(空)'} 不存在`, kind: 'item-not-found' }
    }
    // 操作之目標 id 亦可為同一差量新項之「@ref」(新增處理在前,此時已配發;如把本批新主張 contest 進既有爭議——真實模型驗收實測);
    //   所依之新項未成立 → 依賴失敗(dep＝寫明根因之拒收理由)
    const targetOf = (x) => {
        const v = str(x)
        if (!v.startsWith('@')) return { hit: findItem(s, v), shown: v, dep: '' }
        const r = resolveRef(v)
        return { hit: r.id ? findItem(s, r.id) : null, shown: v, dep: r.dep || '' }
    }
    // 引用須為指定種類(之非終態)項:不成立時回寫明根因之理由——依賴失敗、ref 未定義、不存在、種類不符、終態各自寫明,
    //   不以「不存在或為終態」概括(看不出錯在種類)
    //   依賴失敗另回 dep(各引用點原樣使用,不加前綴——「所引 @x 已被終審剔除(理由)」於全部引用點同一句)
    const refOfKind = (x, keys, opt = {}) => {
        const r = resolveRef(x)
        if (!r.id) return { id: null, why: r.dep || r.reason, dep: r.dep || '', kind: r.kind }
        const h = findItem(s, r.id)
        const shown = str(x) === r.id ? r.id : `${str(x)}＝${r.id}`
        if (!keys.includes(h.key)) return { id: null, why: `${shown} 為${NAME_OF_KEY[h.key] || h.key}，須為${keys.map((k) => NAME_OF_KEY[k] || k).join('或')}`, kind: 'target-kind-mismatch' }
        if (opt.live !== false && TERMINAL_SET.has(h.item.status)) return { id: null, why: `${shown} 已${h.item.status === 'retracted' ? '撤回' : '取代'}`, kind: 'target-terminal' }
        return { id: r.id, hit: h }
    }
    // 操作之目標(id)須為指定種類:不成立時回寫明根因之理由——依賴失敗(dep)、ref 未定義、不存在、種類不符;終態由各操作依其語意自判
    const targetOfKind = (x, keys, noun = '目標') => {
        const tg = targetOf(x)
        if (!tg.hit) {
            const v = str(x)
            const undef = v.startsWith('@') && !refState.has(v.slice(1))
            return { hit: null, why: tg.dep || (undef ? `ref「${v}」未定義` : `${noun} ${tg.shown || '(空)'} 不存在`), kind: tg.dep ? 'dependency-rejected' : (undef ? 'ref-undefined' : 'item-not-found') }
        }
        if (!keys.includes(tg.hit.key)) {
            const shown = str(x) === tg.hit.item.id ? tg.hit.item.id : `${str(x)}＝${tg.hit.item.id}`
            return { hit: null, why: `目標 ${shown} 為${NAME_OF_KEY[tg.hit.key]}，須為${keys.map((k) => NAME_OF_KEY[k]).join('或')}`, kind: 'target-kind-mismatch' }
        }
        return { hit: tg.hit, why: '', kind: '' }
    }
    const sideRange = (d) => ((d.sides || []).length ? `現有 side 0～${d.sides.length - 1}` : '無任何一方')
    const hist = (item, op, note, before) => {
        item.history = isarr(item.history) ? item.history : []
        item.history.push({ version, at, op, ...(note ? { note } : {}), ...(before ? { before } : {}) })
    }
    // 文字中之引用:〔id〕須存在;〔@ref〕須為本差量宣告之 ref(落盤前換成實際 id;所指新項被剔除或拒收者去掉標記,見下方 linkCodes)。
    //   文字中之提及不是結構引用:所指新項不成立時只去掉標記、不連帶拒收本操作——與「同差量稍後才失敗者去掉標記」同一處置
    const danglingIn = (v) => {
        for (const m of v.matchAll(REF_RE)) if (!findItem(s, m[1])) return `文字引用懸空：〔${m[1]}〕`
        for (const m of v.matchAll(TEMP_REF_RE)) if (!refState.has(m[1])) return `文字引用懸空：〔@${m[1]}〕（ref 未定義）`
        return ''
    }
    // 文字欄:長度(逾 2 倍拒收、逾上限計數)與文字中之引用;fields 未給即取該種類之全部文字欄
    //   回傳 { why, kind }(不合)或 ''(合格)
    const textProblem = (key, obj, fields) => {
        for (const f of fields || TEXT_FIELDS[key] || []) {
            const vals = ARRAY_FIELDS.has(f) ? strs(obj[f]) : [str(obj[f])]
            for (const v of vals) {
                const lp = overLength(f, v, limits)
                if (lp) return { why: lp, kind: 'text-too-long' }
                const dg = danglingIn(v)
                if (dg) return { why: dg, kind: 'dangling-text-ref' }
            }
        }
        return ''
    }
    // 逾建議上限(1～2 倍、放行者)之欄數與受檢欄數:比率供外部監測超長率
    const countOver = (key, obj) => {
        for (const f of TEXT_FIELDS[key] || []) {
            const vals = (ARRAY_FIELDS.has(f) ? strs(obj[f]) : [str(obj[f])]).filter(Boolean)
            if (!limits[f] || !vals.length) continue
            lengthChecked++
            if (vals.some((v) => v.length > limits[f])) overLimit++
        }
    }
    const markRef = (ref, id) => {
        if (ref && refState.has(ref)) refState.get(ref).id = id
    }
    const failRef = (op, why) => {
        const ref = str(op?.ref)
        if (ref && refState.has(ref)) Object.assign(refState.get(ref), { rejected: true, why: str(why) })
    }
    const live = (hit) => hit && !TERMINAL_SET.has(hit.item.status)
    // 同篇成對(檔頭④⑬;與同篇之單方判準對稱):兩方之篇集(本方出處 ∪ 所引主張之出處)皆恰為同一篇、且兩方皆非轉述 → 同一研究自身之正反,
    //   不成爭議。只擋「本操作新增之同篇成對」(前後比對;方之索引只增不重排):新立爭議之全部方、補方(sides_add)、併爭議(merge)。
    //   contest／confirm 帶 side／合併主張只使某方之篇集變大,不會新增同篇成對。兩方篇集相同而各 ≥2 篇者不在此列(辯論型報導兩篇皆含兩方——正式庫實例)
    const reported = isarr(ctx.reportedBasis) ? ctx.reportedBasis : REPORTED_BASIS
    const sameDocPairsOf = (sides) => {
        const docs = sides.map((sd) => uniq([...strs(sd?.sources), ...strs(sd?.claims).flatMap((cid) => findItem(s, cid)?.item?.sources || [])]))
        const rep = sides.map((sd) => reported.some((k) => str(sd?.basis).includes(k)))
        const out = []
        for (let i = 0; i < sides.length; i++) for (let j = i + 1; j < sides.length; j++) if (docs[i].length === 1 && docs[j].length === 1 && docs[i][0] === docs[j][0] && !rep[i] && !rep[j]) out.push([i, j])
        return out
    }
    const sameDocWhy = (beforeSides, afterSides) => {
        const prev = new Set(sameDocPairsOf(beforeSides).map((p) => p.join('-')))
        const added = sameDocPairsOf(afterSides).find((p) => !prev.has(p.join('-')))
        return added ? `side ${added[0]} 與 side ${added[1]} 之出處只有同一篇（同篇正反不成爭議；引述他人研究之一方請標 basis「${reported[0]}」）` : ''
    }
    /**
     * 本質引用之單一處置(檔頭⑭):fromId 自本質所引移除;toId 為存活主張時改指之,否則標 stale(本質待重寫)
     *
     * @param {String} fromId 輸入轉終態之主張 id
     * @param {String|null} toId 輸入改指目標 id(null＝只移除)
     */
    const redirectEssence = (fromId, toId) => {
        const ess = s.essence
        if (!isarr(ess?.claims) || !ess.claims.includes(fromId)) return
        const h = toId ? findItem(s, toId) : null
        const ok = !!(h && h.key === 'claims' && !TERMINAL_SET.has(h.item.status))
        ess.claims = uniq(ess.claims.map((x) => (x === fromId ? (ok ? toId : null) : x)).filter(Boolean))
        if (!ok) ess.stale = { version, at, from: fromId }
    }
    let essenceSeen = false
    // 終審存疑之落帳:爭議上留最新一則(含累計次數),另寫沿革(沿革只增,先前之存疑可追溯)
    const markDoubt = (op, entry) => {
        const did = op.op === 'contest' ? resolveRef(op.dispute).id : entry.id
        const d = did ? findItem(s, did)?.item : null
        if (!d || !isarr(d.sides)) return
        const note = str(op._doubt.note).slice(0, (limits.note || 120) * 2) || '（終審未附說明）'
        const prev = isobj(d.doubt) ? d.doubt : null
        d.doubt = { reason: str(op._doubt.reason) || '非對立', note, version, at, op: op.op, ...(op.op === 'contest' ? { claim: entry.id, side: parseSide(op.side) } : {}), count: (prev?.count || 0) + 1 }
        hist(d, 'doubt', `${d.doubt.reason}：${note}`)
        touched.add(d.id)
    }

    for (const { op, index } of order) {
        const fail = (reason, kind) => {
            failRef(op, reason)
            reject(index, op, reason, kind)
        }
        if (!isobj(op) || !allowed.includes(op.op)) {
            const known = isobj(op) && ALL_OPS.has(op.op)
            const outOfPass = known && only && !only.includes(op.op)
            fail(known ? (outOfPass ? `本 pass 只收：${only.join('、')}（收到「${op.op}」）` : `${{ consolidate: '整併', delta: '增量' }[mode]}模式不可用「${op.op}」`) : '未知操作', known ? (outOfPass ? 'op-not-in-pass' : 'op-not-in-mode') : 'op-unknown')
            continue
        }
        const ref = str(op.ref)
        if (ref && refDup.has(ref)) {
            fail(`ref「${ref}」重複`, 'ref-duplicate')
            continue
        }
        if (conflict.has(index)) {
            fail(conflict.get(index).why, conflict.get(index).kind)
            continue
        }
        const guardWhy = launderOf(op) || recheckRetractOf(op)
        if (guardWhy) {
            fail(guardWhy.why, guardWhy.kind)
            continue
        }
        if (withhold.has(op.op) || op._withhold === true) {
            failRef(op)
            // 暫緩之理由(_withholdReason;如出處支持覆核未完成)與是否為「覆核判不了」(_withholdEs9;待審之 es9Tries 計次)隨之回報,存待審時取用
            withheld.push({ index: rep(index, op), op: op.op, sources: opSources(op), _op: op, why: str(op._withholdReason), es9: op._withholdEs9 === true })
            continue
        }
        const needSrc = (v) => {
            const got = srcOf(v)
            return got.length ? got : null
        }
        const done = (entry) => applied.push({ index: rep(index, op), op: op.op, sources: [], ...entry })
        const nApplied = applied.length

        switch (op.op) {
        case 'add': {
            const kind = str(op.kind || op.type)
            if (!kinds.includes(kind)) {
                fail(`種類「${kind}」不在許可清單（${kinds.join('、')}）`, 'kind-not-allowed')
                break
            }
            if (!str(op.text)) {
                fail('缺 text', 'missing-field')
                break
            }
            const src = needSrc(op.sources)
            if (!src) {
                fail('無本批出處', 'no-batch-source')
                break
            }
            const tp = textProblem('claims', op)
            if (tp) {
                fail(tp.why, tp.kind)
                break
            }
            // 與既有(非終態)主張全文相同 → 改記為 confirm
            const dup = s.claims.find((c) => !TERMINAL_SET.has(c.status) && sameText(c.text) === sameText(op.text))
            if (dup) {
                const add = src.filter((x) => !dup.sources.includes(x))
                dup.sources = uniq([...dup.sources, ...src])
                if (add.length) hist(dup, 'confirm', 'add 與既有主張同文，改記為確認')
                touched.add(dup.id)
                markRef(ref, dup.id)
                done({ id: dup.id, sources: src, asConfirm: true, noop: add.length === 0 })
                break
            }
            const id = newId('claims')
            markRef(ref, id)
            const c = {
                id,
                kind,
                facet: str(op.facet),
                text: str(op.text),
                conditions: strs(op.conditions),
                pros: strs(op.pros),
                cons: strs(op.cons),
                period: str(op.period),
                critique: str(op.critique),
                basis: str(op.basis),
                sources: src,
                origin: 'note',
                status: 'active',
                statusNote: '',
                validPeriod: '',
                supersededBy: '',
                mergedInto: '',
                retractReason: '',
                evidence: { level: '', basis: '', trace: '' },
                addedIn: version,
                history: [{ version, at, op: 'add' }],
            }
            countOver('claims', c)
            s.claims.push(c)
            touched.add(id)
            done({ id, sources: src })
            break
        }
        case 'confirm': {
            const tk = targetOfKind(op.id, ['claims', 'parameters', 'disputes'])
            const hit = tk.hit
            if (!hit) {
                fail(tk.why, tk.kind)
                break
            }
            if (!live(hit)) {
                fail(`目標 ${hit.item.id} 已${hit.item.status === 'retracted' ? '撤回' : '取代'}`, 'target-terminal')
                break
            }
            const src = needSrc(op.sources)
            if (!src) {
                fail('無本批出處', 'no-batch-source')
                break
            }
            const it = hit.item
            if (hit.key === 'disputes') {
                // 確認爭議:出處加到指定之一方(side 索引,0 起),未指定者加到爭議本身
                const given = op.side !== undefined && op.side !== null && op.side !== ''
                const k = given ? parseSide(op.side) : null
                const side = k === null ? null : it.sides[k]
                if (given && !side) {
                    fail(`爭議 ${it.id} 無 side ${str(op.side)}（${sideRange(it)}）`, 'side-invalid')
                    break
                }
                if (side) side.sources = uniq([...(side.sources || []), ...src])
            }
            const add = src.filter((x) => !(it.sources || []).includes(x))
            it.sources = uniq([...(it.sources || []), ...src])
            if (add.length) hist(it, 'confirm')
            touched.add(it.id)
            done({ id: it.id, sources: src, noop: add.length === 0 })
            break
        }
        case 'revise': {
            const tk = targetOfKind(op.id, Object.keys(REVISABLE))
            const hit = tk.hit
            if (!hit) {
                fail(tk.why, tk.kind)
                break
            }
            if (!live(hit)) {
                fail(`目標 ${hit.item.id} 為終態，不可修訂`, 'target-terminal')
                break
            }
            if (!str(op.reason)) {
                fail('缺 reason', 'missing-field')
                break
            }
            let src = []
            if (mode === 'delta' && !program) {
                src = needSrc(op.sources)
                if (!src) {
                    fail('無本批出處', 'no-batch-source')
                    break
                }
            }
            else src = srcOf(op.sources) // 結構重組(整併、複審)與程式差量(⑱:依出處收斂措辭)可不帶出處;帶了則累積
            const f = isobj(op.fields) ? op.fields : {}
            const keys = Object.keys(f).filter((k) => REVISABLE[hit.key].includes(k))
            if (!keys.length) {
                fail('無可改之欄位', 'revise-no-editable-field')
                break
            }
            if (keys.includes('kind') && !kinds.includes(str(f.kind))) {
                fail(`種類「${str(f.kind)}」不在許可清單`, 'kind-not-allowed')
                break
            }
            const next = {}
            // 參數之 conditions 為字串(與 param_add 同一形狀);主張之 conditions、pros、cons 為陣列
            for (const k of keys) next[k] = k === 'snapshot' ? f[k] === true : (hit.key === 'parameters' && k === 'conditions' ? paramConditions(f[k]) : (ARRAY_FIELDS.has(k) ? strs(f[k]) : str(f[k])))
            if ((hit.key === 'claims' && keys.includes('text') && !next.text) || (hit.key === 'parameters' && ((keys.includes('name') && !next.name) || (keys.includes('value') && !next.value)))) {
                fail('必要欄位不可改為空', 'required-field-emptied')
                break
            }
            // ⑩只驗被改之欄位
            const tp = textProblem(hit.key, next, keys.filter((k) => (TEXT_FIELDS[hit.key] || []).includes(k)))
            if (tp) {
                fail(tp.why, tp.kind)
                break
            }
            const it = hit.item
            const before = {}
            for (const k of keys) before[k] = clone(it[k] ?? '')
            Object.assign(it, next)
            it.sources = uniq([...(it.sources || []), ...src])
            hist(it, 'revise', str(op.reason), before)
            countOver(hit.key, it)
            touched.add(it.id)
            done({ id: it.id, sources: src })
            break
        }
        case 'supersede': {
            const tk = targetOfKind(op.id, ['claims', 'parameters'])
            const hit = tk.hit
            if (!hit) {
                fail(tk.why, tk.kind)
                break
            }
            if (!live(hit)) {
                fail(`目標 ${hit.item.id} 已為終態`, 'target-terminal')
                break
            }
            if (!str(op.reason)) {
                fail('缺 reason', 'missing-field')
                break
            }
            if (!str(op.validPeriod)) {
                fail('缺 validPeriod（舊觀點成立期間；年份不詳可寫「較早之研究」或「不詳」）', 'missing-field')
                break
            }
            const src = needSrc(op.sources)
            if (!src) {
                fail('無本批出處', 'no-batch-source')
                break
            }
            let by = ''
            if (str(op.by)) {
                const r = refOfKind(op.by, [hit.key], { live: false })
                if (!r.id || r.id === hit.item.id) {
                    fail(r.id ? '取代者不可為自身' : (r.dep || `取代者 ${r.why}`), r.id ? 'supersede-by-self' : (r.dep ? 'dependency-rejected' : r.kind))
                    break
                }
                by = r.id
            }
            const it = hit.item
            Object.assign(it, { status: 'superseded', statusNote: str(op.reason), validPeriod: str(op.validPeriod), supersededBy: by })
            it.sources = uniq([...(it.sources || []), ...src])
            hist(it, 'supersede', str(op.reason))
            // 本質所引者移除並標待重寫(不改指 by:本質文字仍述舊觀點,改指新主張即假支持;檔頭⑭)
            if (hit.key === 'claims') redirectEssence(it.id, null)
            touched.add(it.id)
            done({ id: it.id, sources: src })
            break
        }
        case 'retract': {
            const tk = targetOfKind(op.id, Object.keys(PREFIX_OF_KEY))
            const hit = tk.hit
            if (!hit) {
                fail(tk.why, tk.kind)
                break
            }
            if (TERMINAL_SET.has(hit.item.status)) {
                fail(`目標 ${hit.item.id} 已為終態`, 'target-terminal')
                break
            }
            const reason = str(op.reason)
            if (!retractReasons.includes(reason)) {
                fail(`撤回理由須為：${retractReasons.join('、')}`, 'retract-reason-invalid')
                break
            }
            const src = srcOf(op.sources)
            const it = hit.item
            // 「重複」須指明存活之同種類項,出處併入之(否則存活者得不到被撤回者之出處,出處累積失真)
            let into = null
            if (reason === '重複') {
                const r = resolveRef(op.into)
                const h = r.id ? findItem(s, r.id) : null
                if (!h || h.key !== hit.key || h.item.id === it.id || !live(h)) {
                    fail(r.dep || '理由「重複」須帶 into（存活之同種類項）', r.dep ? 'dependency-rejected' : 'retract-duplicate-needs-into')
                    break
                }
                into = h.item
            }
            it.status = 'retracted'
            it.retractReason = reason
            it.statusNote = str(op.note) || (into ? `與 ${into.id} 重複` : reason)
            if (into) {
                it.mergedInto = into.id
                into.sources = uniq([...(into.sources || []), ...(it.sources || [])])
                hist(into, 'retract', `併入重複項 ${it.id} 之出處`)
                touched.add(into.id)
            }
            hist(it, 'retract', it.statusNote)
            // 本質所引者:「重複」改指存活之 into,其餘移除並標待重寫(檔頭⑭)
            if (hit.key === 'claims') redirectEssence(it.id, into ? into.id : null)
            touched.add(it.id)
            done({ id: it.id, sources: src })
            break
        }
        case 'contest': {
            const tk = targetOfKind(op.id, ['claims'], '目標主張')
            const hit = tk.hit
            if (!hit) {
                fail(tk.why, tk.kind)
                break
            }
            if (!live(hit)) {
                fail(`目標 ${hit.item.id} 已為終態`, 'target-terminal')
                break
            }
            if (!str(op.reason)) {
                fail('缺 reason', 'missing-field')
                break
            }
            const r = refOfKind(op.dispute, ['disputes'], { live: false })
            if (!r.id) {
                fail(!str(op.dispute) ? 'contest 須連既有或同一差量新建之爭議' : (r.dep || `所連爭議 ${r.why}`), !str(op.dispute) ? 'contest-needs-dispute' : (r.dep ? 'dependency-rejected' : r.kind))
                break
            }
            const d = findItem(s, r.id).item
            if (d.status !== 'open') {
                fail(`爭議 ${d.id} 已非未解決`, 'target-not-open')
                break
            }
            const k = parseSide(op.side)
            const side = k === null ? null : d.sides[k]
            if (!side) {
                fail(`爭議 ${d.id} 無 side ${str(op.side) || '(未給)'}（contest 須指明 side；${sideRange(d)}）`, 'side-invalid')
                break
            }
            const src = needSrc(op.sources)
            if (!src) {
                fail('無本批出處', 'no-batch-source')
                break
            }
            side.claims = uniq([...(side.claims || []), hit.item.id])
            side.sources = uniq([...(side.sources || []), ...src])
            d.sources = uniq([...(d.sources || []), ...src])
            hist(hit.item, 'contest', `${d.id}：${str(op.reason)}`)
            hist(d, 'contest', `${hit.item.id} 列入 side ${k}`)
            touched.add(hit.item.id)
            touched.add(d.id)
            done({ id: hit.item.id, sources: src })
            break
        }
        case 'dispute_add':
        case 'dispute_update': {
            const isAdd = op.op === 'dispute_add'
            let d = null
            if (!isAdd) {
                const tk = targetOfKind(op.id, ['disputes'], '目標爭議')
                const hit = tk.hit
                if (!hit) {
                    fail(tk.why, tk.kind)
                    break
                }
                if (TERMINAL_SET.has(hit.item.status)) {
                    fail(`爭議 ${hit.item.id} 已撤回`, 'target-terminal')
                    break
                }
                d = hit.item
                if (!str(op.note)) {
                    fail('缺 note', 'missing-field')
                    break
                }
                if (op.status !== undefined && !['open', 'resolved'].includes(op.status)) {
                    fail('status 須為 open／resolved', 'dispute-status-invalid')
                    break
                }
            }
            else if (!str(op.question)) {
                fail('缺 question', 'missing-field')
                break
            }
            const rawSides = isAdd ? (isarr(op.sides) ? op.sides : []) : (isarr(op.sides_add) ? op.sides_add : [])
            if (isAdd && rawSides.length < 2) {
                fail('少於兩方', 'dispute-sides-too-few')
                break
            }
            const sides = []
            let bad = ''
            for (const sd of rawSides) {
                if (!str(sd?.position)) {
                    bad = { why: '有一方缺立場', kind: 'dispute-side-no-position' }
                    break
                }
                const claims = []
                for (const x of strs(sd?.claims)) {
                    const r = refOfKind(x, ['claims'])
                    if (!r.id) {
                        bad = { why: r.why, kind: r.kind }
                        break
                    }
                    claims.push(r.id)
                }
                if (bad) break
                const src = srcOf(sd?.sources)
                if (!src.length && !claims.length) {
                    bad = { why: '有一方無本批出處亦無所引主張', kind: 'dispute-side-no-source' }
                    break
                }
                const tp = textProblem('disputes', sd, ['position', 'conditions', 'basis'])
                if (tp) {
                    bad = tp
                    break
                }
                sides.push({ position: str(sd.position), conditions: strs(sd?.conditions), claims, sources: src, ...(str(sd?.basis) ? { basis: str(sd.basis) } : {}) })
            }
            if (bad) {
                fail(bad.why, bad.kind)
                break
            }
            const newSrc = uniq([...srcOf(op.sources), ...sides.flatMap((x) => x.sources)])
            if (!newSrc.length) {
                fail('無本批出處', 'no-batch-source')
                break
            }
            const tp = textProblem('disputes', isAdd ? { question: op.question } : { note: op.note })
            if (tp) {
                fail(tp.why, tp.kind)
                break
            }
            // ④同篇正反不成爭議(⑬轉述之方例外):新立之全部方、補方與既有各方之間,不得新增同篇成對
            const sameDoc = sameDocWhy(isAdd ? [] : d.sides, isAdd ? sides : [...d.sides, ...sides])
            if (sameDoc) {
                fail(sameDoc, 'same-doc-pair')
                break
            }
            if (isAdd) {
                const id = newId('disputes')
                markRef(ref, id)
                d = { id, question: str(op.question), sides, sources: newSrc, status: 'open', note: str(op.note), origin: 'note', evidence: { level: '', basis: '', trace: '' }, addedIn: version, history: [{ version, at, op: 'dispute_add' }] }
                s.disputes.push(d)
            }
            else {
                d.sides.push(...sides)
                if (op.status) d.status = op.status
                d.note = str(op.note)
                d.sources = uniq([...(d.sources || []), ...newSrc])
                hist(d, 'dispute_update', str(op.note))
            }
            touched.add(d.id)
            for (const sd of sides) for (const cid of sd.claims) touched.add(cid)
            done({ id: d.id, sources: newSrc })
            break
        }
        case 'dispute_dissolve': {
            // 拆解硬配之對立(stores/coreState 檔頭⑧):爭議轉終態,只有立場文字之方轉為獨立主張——內容與出處全留,只拿掉對立標籤
            const tk = targetOfKind(op.id, ['disputes'], '目標爭議')
            const hit = tk.hit
            if (!hit) {
                fail(tk.why, tk.kind)
                break
            }
            if (TERMINAL_SET.has(hit.item.status)) {
                fail(`爭議 ${hit.item.id} 已撤回`, 'target-terminal')
                break
            }
            const reason = str(op.reason)
            if (!reason) {
                fail('缺 reason（兩方為何不是對立）', 'missing-field')
                break
            }
            const tp = textProblem('disputes', { note: reason })
            if (tp) {
                fail(tp.why, tp.kind)
                break
            }
            const d = hit.item
            // 各方之去處(內容與出處全留):只有立場文字之方 → 轉為主張;引主張之方其「方層出處」(confirm side／contest 所加,
            //   不在所引主張出處內者)→ 只引一條有效主張則併入其出處,否則以該方立場另立一條主張——拆解後這些出處仍被有效項引用
            //   種類取 kinds[k](k＝side 索引;須在主張種類之清單,否則整條拒收——與 ES14 之組裝同一判定;種類可由 domain 自訂);標注取 annotations[k]
            //   ({ critique, pros, cons, period, conditions };條件與該方之 conditions 合併——轉為 pitfall 而該方無條件時由此補;
            //   檢定標準 ES4——拆出之主張與新增者同一標準)
            const kindsIn = isarr(op.kinds) ? op.kinds.map((x) => str(x)) : []
            const annIn = isarr(op.annotations) ? op.annotations : []
            const plan = []
            const merges = []
            let bad = ''
            for (const { k, to, cited, extra } of dissolveSidesOf(s, d)) {
                const sd = d.sides[k]
                if (to === 'none') continue
                if (to === 'merge') {
                    merges.push({ k, claim: cited[0].item, extra })
                    continue
                }
                const ann = isobj(annIn[k]) ? annIn[k] : {}
                const c0 = { text: str(sd.position), conditions: uniq([...strs(sd.conditions), ...strs(ann.conditions)]), critique: str(ann.critique), pros: strs(ann.pros), cons: strs(ann.cons), period: str(ann.period) }
                const src = (sd.claims || []).length ? extra : (sd.sources || [])
                const p = textProblem('claims', c0, ['text', 'conditions', 'critique', 'pros', 'cons', 'period'])
                if (p || !src.length) {
                    bad = { why: `side ${k} 無法轉為主張：${p ? p.why : '無出處'}`, kind: 'dissolve-side-unconvertible' }
                    break
                }
                if (!kinds.includes(kindsIn[k])) {
                    bad = { why: `side ${k} 之種類「${kindsIn[k] || '（未給）'}」不在主張種類（${kinds.join('、')}）`, kind: 'kind-not-allowed' }
                    break
                }
                plan.push({ k, sd, c0, src, kind: kindsIn[k] })
            }
            if (bad) {
                fail(bad.why, bad.kind)
                break
            }
            for (const m of merges) {
                m.claim.sources = uniq([...(m.claim.sources || []), ...m.extra])
                hist(m.claim, 'dispute_dissolve', `併入 ${d.id} side ${m.k} 之方層出處`)
                touched.add(m.claim.id)
            }
            for (const x of plan) {
                const id = newId('claims')
                const c = { id, kind: x.kind, facet: '', text: x.c0.text, conditions: x.c0.conditions, pros: x.c0.pros, cons: x.c0.cons, period: x.c0.period, critique: x.c0.critique, basis: str(x.sd.basis), sources: [...x.src], origin: 'dispute', status: 'active', statusNote: '', validPeriod: '', supersededBy: '', mergedInto: '', retractReason: '', evidence: { level: '', basis: '', trace: '' }, addedIn: version, history: [{ version, at, op: 'dispute_dissolve', note: `拆自 ${d.id} side ${x.k}` }] }
                countOver('claims', c)
                s.claims.push(c)
                x.sd.claims = uniq([...(x.sd.claims || []), id])
                touched.add(id)
            }
            Object.assign(d, { status: 'retracted', retractReason: DISSOLVED_RETRACT_REASON, statusNote: reason })
            hist(d, 'dispute_dissolve', reason)
            touched.add(d.id)
            for (const sd of d.sides || []) for (const cid of sd.claims || []) touched.add(cid)
            done({ id: d.id })
            break
        }
        case 'param_add': {
            if (!str(op.name) || !str(op.value)) {
                fail('缺 name 或 value', 'missing-field')
                break
            }
            const src = needSrc(op.sources)
            if (!src) {
                fail('無本批出處', 'no-batch-source')
                break
            }
            let claim = ''
            if (str(op.claim)) {
                const r = refOfKind(op.claim, ['claims'], { live: false })
                if (!r.id) {
                    fail(r.dep || `所依主張 ${r.why}`, r.dep ? 'dependency-rejected' : r.kind)
                    break
                }
                claim = r.id
            }
            const p0 = { name: op.name, value: op.value, conditions: op.conditions, period: op.period }
            const tp = textProblem('parameters', p0)
            if (tp) {
                fail(tp.why, tp.kind)
                break
            }
            const id = newId('parameters')
            markRef(ref, id)
            // 參數之 conditions 為字串(渲染之表格與狀態摘要以字串呈現);模型給陣列(同主張之形狀)者以「；」併為一句——否則陣列被當成空字串,條件靜默遺失
            const p = { id, name: str(op.name), value: str(op.value), conditions: paramConditions(op.conditions), period: str(op.period), snapshot: op.snapshot === true, claim, sources: src, origin: 'note', status: 'active', statusNote: '', validPeriod: '', supersededBy: '', retractReason: '', evidence: { level: '', basis: '', trace: '' }, addedIn: version, history: [{ version, at, op: 'param_add' }] }
            countOver('parameters', p)
            s.parameters.push(p)
            touched.add(id)
            done({ id, sources: src })
            break
        }
        case 'question_add': {
            if (!str(op.text)) {
                fail('缺 text', 'missing-field')
                break
            }
            const tp = textProblem('questions', { text: op.text })
            if (tp) {
                fail(tp.why, tp.kind)
                break
            }
            const src = srcOf(op.sources)
            const id = newId('questions')
            markRef(ref, id)
            s.questions.push({ id, text: str(op.text), status: 'open', resolution: '', sources: src, retractReason: '', addedIn: version, history: [{ version, at, op: 'question_add' }] })
            touched.add(id)
            done({ id, sources: src })
            break
        }
        case 'question_resolve': {
            const tk = targetOfKind(op.id, ['questions'], '目標問題')
            const hit = tk.hit
            if (!hit) {
                fail(tk.why, tk.kind)
                break
            }
            if (hit.item.status !== 'open') {
                fail(`問題 ${hit.item.id} 已非未解`, 'target-not-open')
                break
            }
            if (!str(op.resolution)) {
                fail('缺 resolution', 'missing-field')
                break
            }
            const src = needSrc(op.sources)
            if (!src) {
                fail('無本批出處', 'no-batch-source')
                break
            }
            const tp = textProblem('questions', { resolution: op.resolution })
            if (tp) {
                fail(tp.why, tp.kind)
                break
            }
            Object.assign(hit.item, { status: 'resolved', resolution: str(op.resolution), sources: uniq([...hit.item.sources, ...src]) })
            hist(hit.item, 'question_resolve')
            touched.add(hit.item.id)
            done({ id: hit.item.id, sources: src })
            break
        }
        case 'essence': {
            if (essenceSeen) {
                fail('同一差量只能有一個 essence', 'essence-duplicate')
                break
            }
            const text = str(op.text)
            if (!text) {
                fail('缺 text', 'missing-field')
                break
            }
            // reason 只在改寫既有本質時入沿革(essenceHistory);首次歸納不記錄 reason,故不因缺 reason 整條拒收——
            //   整合員重寫本質操作時可漏 reason,整條拒收即新核心無本質
            if (s.essence.text && !str(op.reason)) {
                fail('缺 reason（改寫本質須說明理由）', 'missing-field')
                break
            }
            // 本質為硬上限(逾即拒,不同於主張各欄之「2 倍拒收、1～2 倍計數」)
            const eo = essenceOver(text, limits)
            if (eo) {
                fail(eo, 'essence-too-long')
                break
            }
            const claims = []
            let why = ''
            for (const x of strs(op.claims)) {
                const r = refOfKind(x, ['claims'])
                if (!r.id) {
                    why = r
                    break
                }
                claims.push(r.id)
            }
            if (why) {
                fail(why.why, why.kind)
                break
            }
            if (!claims.length) {
                fail('essence 須引至少一條主張（claims）', 'essence-claims-required')
                break
            }
            const dang = danglingIn(text)
            if (dang) {
                fail(dang, 'dangling-text-ref')
                break
            }
            essenceSeen = true
            if (s.essence.text) s.essenceHistory.push({ version: s.version, at: s.essence.at || '', text: s.essence.text, reason: str(op.reason) })
            s.essence = { text, claims: uniq(claims), sources: [], at, stale: null } // 重寫即解除待重寫標記
            done({ id: 'essence' })
            break
        }
        case 'related_add':
        case 'related_prune': {
            const cs = strs(op.concepts)
            if (!cs.length) {
                fail('缺 concepts', 'missing-field')
                break
            }
            const selfKey = keyOf(ctx.self)
            if (op.op === 'related_add') {
                const have = new Set(s.related.map(keyOf))
                for (const c of cs) {
                    const k = keyOf(c)
                    if (!k || k === selfKey || have.has(k)) continue
                    have.add(k)
                    s.related.push(c)
                }
            }
            else {
                const drop = new Set(cs.map(keyOf))
                s.related = s.related.filter((c) => !drop.has(keyOf(c)))
            }
            done({})
            break
        }
        case 'merge': {
            const ri = refOfKind(op.into, ['claims', 'parameters', 'questions', 'disputes'])
            if (!ri.id) {
                fail(ri.dep || `into ${ri.why}`, ri.dep ? 'dependency-rejected' : ri.kind)
                break
            }
            const into = ri.hit
            if (!str(op.reason)) {
                fail('缺 reason', 'missing-field')
                break
            }
            const from = uniq(strs(op.from))
            if (!from.length || from.includes(into.item.id)) {
                fail('from 為空或含 into', 'merge-invalid-from')
                break
            }
            const badFrom = from.map((id) => refOfKind(id, [into.key])).find((r) => !r.id)
            if (badFrom) {
                fail(badFrom.dep || `from ${badFrom.why}`, badFrom.dep ? 'dependency-rejected' : badFrom.kind)
                break
            }
            const hits = from.map((id) => findItem(s, id))
            // ⑪爭議併爭議:各方併入 into(方數只增),被併之爭議轉終態(重複,mergedInto);併後不得新增同篇成對(④)
            if (into.key === 'disputes') {
                const sameDoc = sameDocWhy(into.item.sides || [], [...(into.item.sides || []), ...hits.flatMap((h) => h.item.sides || [])])
                if (sameDoc) {
                    fail(`併後${sameDoc}`, 'same-doc-pair')
                    break
                }
                for (const h of hits) {
                    into.item.sides = [...(into.item.sides || []), ...clone(h.item.sides || [])]
                    into.item.sources = uniq([...(into.item.sources || []), ...(h.item.sources || [])])
                    Object.assign(h.item, { status: 'retracted', retractReason: '重複', statusNote: `併入 ${into.item.id}：${str(op.reason)}`, mergedInto: into.item.id })
                    hist(h.item, 'merge', `併入 ${into.item.id}`)
                    touched.add(h.item.id)
                }
                hist(into.item, 'merge', `併入 ${from.join('、')}：${str(op.reason)}`)
                touched.add(into.item.id)
                for (const sd of into.item.sides || []) for (const cid of sd.claims || []) touched.add(cid)
                done({ id: into.item.id })
                break
            }
            if (str(op.text) && into.key === 'claims') {
                // ⑪合併稿須保有 into 與每條被併項之過半內容(防合併時把既有知識改寫掉);長度上限放寬到 max(上限×2, 最長輸入×1.2)
                const textOf = (it) => str(it.text)
                const inputs = [into.item, ...hits.map((h) => h.item)]
                const lost = inputs.filter((it) => containment(textOf(it), op.text) < MERGE_KEEP_MIN).map((it) => it.id)
                if (lost.length) {
                    fail(`合併稿未保有 ${lost.join('、')} 之過半內容（包含率須 ≥${MERGE_KEEP_MIN}）；不改寫文字可省略 text`, 'merge-text-overwrite')
                    break
                }
                const cap = Math.max((limits.text || 90) * 2, Math.ceil(1.2 * Math.max(...inputs.map((it) => textOf(it).length))))
                if (str(op.text).length > cap) {
                    fail(`text 過長（${str(op.text).length} 字，上限 ${cap}）`, 'text-too-long')
                    break
                }
                const dg = danglingIn(str(op.text))
                if (dg) {
                    fail(dg, 'dangling-text-ref')
                    break
                }
            }
            // 併入會使同一爭議之不同方指向同一主張者拒收(兩方立場相反卻引同一條;爭議兩方所引之近似主張不是同義)
            if (into.key === 'claims') {
                const ids = new Set([into.item.id, ...from])
                const clash = s.disputes.find((d) => !TERMINAL_SET.has(d.status) && new Set((d.sides || []).map((sd, k) => ((sd.claims || []).some((x) => ids.has(x)) ? k : -1)).filter((k) => k >= 0)).size > 1)
                if (clash) {
                    fail(`合併會使爭議 ${clash.id} 之不同方指向同一主張（${[...ids].join('、')} 分屬其各方）`, 'merge-dispute-clash')
                    break
                }
            }
            for (const h of hits) {
                Object.assign(h.item, { status: 'superseded', statusNote: `併入 ${into.item.id}：${str(op.reason)}`, mergedInto: into.item.id, supersededBy: into.item.id })
                hist(h.item, 'merge', `併入 ${into.item.id}`)
                into.item.sources = uniq([...(into.item.sources || []), ...(h.item.sources || [])])
                touched.add(h.item.id)
            }
            // 被併之主張若被爭議某方或本質引用,改指向 into(引用不懸空、contested 衍生正確)
            if (into.key === 'claims') {
                const map = new Set(from)
                for (const d of s.disputes) {
                    for (const sd of d.sides || []) {
                        if ((sd.claims || []).some((x) => map.has(x))) {
                            sd.claims = uniq(sd.claims.map((x) => (map.has(x) ? into.item.id : x)))
                            touched.add(d.id)
                        }
                    }
                }
                for (const f of from) redirectEssence(f, into.item.id)
                for (const p of s.parameters) {
                    if (map.has(p.claim)) {
                        p.claim = into.item.id
                        touched.add(p.id)
                    }
                }
                if (str(op.text)) {
                    hist(into.item, 'merge', str(op.reason), { text: into.item.text })
                    into.item.text = str(op.text)
                    countOver('claims', { text: into.item.text })
                }
            }
            hist(into.item, 'merge', `併入 ${from.join('、')}`)
            touched.add(into.item.id)
            done({ id: into.item.id })
            break
        }
        case 'split': {
            // ⑫長條目拆為多條:各條須為原文子集、合起來涵蓋原文;出處與 origin 沿用原條;原條轉終態(splitInto)——純重組,不需新出處
            const tk = targetOfKind(op.id, ['claims'], '目標主張')
            const hit = tk.hit
            if (!hit) {
                fail(tk.why, tk.kind)
                break
            }
            if (!live(hit)) {
                fail(`目標 ${hit.item.id} 已為終態`, 'target-terminal')
                break
            }
            if (!str(op.reason)) {
                fail('缺 reason', 'missing-field')
                break
            }
            const parts = isarr(op.parts) ? op.parts.filter((p) => isobj(p) && str(p.text)) : []
            if (parts.length < 2 || parts.length > 8) {
                fail('parts 須為 2～8 條（各帶 text）', 'split-parts-invalid')
                break
            }
            const it = hit.item
            // 子集與涵蓋皆以全部文字欄計(text、conditions、pros、cons、critique):只看 text 則其他欄可夾帶原文沒有之內容,
            //   且原條之條件移入拆出條之 conditions 時會被誤判為丟失
            const textOf = (x) => [str(x.text), ...strs(x.conditions), ...strs(x.pros), ...strs(x.cons), str(x.critique)].filter(Boolean).join('\n')
            const orig = textOf(it)
            const weak = parts.map((p, k) => ({ k, c: containment(textOf(p), orig) })).filter((x) => x.c < SPLIT_PART_MIN)
            if (weak.length) {
                fail(`第 ${weak.map((x) => x.k + 1).join('、')} 條不是原文之子集（包含率須 ≥${SPLIT_PART_MIN}）——拆分不可加入原文沒有之內容`, 'split-part-not-subset')
                break
            }
            const cover = containment(orig, parts.map(textOf).join('\n'))
            if (cover < SPLIT_COVER_MIN) {
                fail(`各條合起來只涵蓋原文 ${cover.toFixed(2)}（須 ≥${SPLIT_COVER_MIN}）——拆分不可丟掉原文內容`, 'split-coverage-low')
                break
            }
            let bad = ''
            for (const p of parts) {
                if (str(p.kind) && !kinds.includes(str(p.kind))) {
                    bad = { why: `種類「${str(p.kind)}」不在許可清單`, kind: 'kind-not-allowed' }
                    break
                }
                const tp = textProblem('claims', p, ['text', 'conditions', 'pros', 'cons', 'critique', 'period', 'basis'])
                if (tp) {
                    bad = tp
                    break
                }
            }
            if (bad) {
                fail(bad.why, bad.kind)
                break
            }
            const ids = []
            for (const p of parts) {
                const id = newId('claims')
                const c = {
                    id, kind: str(p.kind) || it.kind, facet: str(p.facet) || it.facet || '', text: str(p.text), conditions: strs(p.conditions), pros: strs(p.pros), cons: strs(p.cons),
                    // 證據性質沿用原條(只拆不改;拆分不得改變證據性質而抬高封頂)
                    period: str(p.period) || str(it.period), critique: str(p.critique), basis: str(it.basis), sources: [...(it.sources || [])],
                    origin: it.origin || 'note',
                    status: 'active', statusNote: '', validPeriod: '', supersededBy: '', mergedInto: '', retractReason: '',
                    evidence: { level: '', basis: '', trace: '' }, addedIn: version, history: [{ version, at, op: 'split', note: `拆自 ${it.id}` }],
                }
                countOver('claims', c)
                s.claims.push(c)
                ids.push(id)
                touched.add(id)
            }
            // 引用原條之爭議方、本質、參數改指第一條(引用不懸空;細部由後續操作調整)
            for (const d of s.disputes) {
                for (const sd of d.sides || []) {
                    if ((sd.claims || []).includes(it.id)) {
                        sd.claims = uniq(sd.claims.map((x) => (x === it.id ? ids[0] : x)))
                        touched.add(d.id)
                    }
                }
            }
            redirectEssence(it.id, ids[0])
            for (const p of s.parameters) {
                if (p.claim === it.id) {
                    p.claim = ids[0]
                    touched.add(p.id)
                }
            }
            Object.assign(it, { status: 'superseded', statusNote: `拆分為 ${ids.join('、')}：${str(op.reason)}`, splitInto: ids })
            hist(it, 'split', `拆分為 ${ids.join('、')}`)
            touched.add(it.id)
            done({ id: it.id, sources: [] })
            break
        }
        case 'refacet': {
            const from = str(op.from)
            const to = str(op.to)
            if (!from || !to || from === to) {
                fail('refacet 須有不同之 from 與 to', 'refacet-invalid')
                break
            }
            const hit = s.claims.filter((c) => !TERMINAL_SET.has(c.status) && c.facet === from)
            if (!hit.length) {
                fail(`無面向為「${from}」之主張`, 'refacet-no-match')
                break
            }
            for (const c of hit) {
                hist(c, 'refacet', `${from} → ${to}`, { facet: from })
                c.facet = to
                touched.add(c.id)
            }
            done({})
            break
        }
        default:
            fail('未知操作', 'op-unknown')
        }
        // 終審存疑(stores/coreState 檔頭⑧):附於爭議類操作、且該操作套用成功者,記於其爭議——不移除、不改狀態;被拒收之操作不記
        if (isobj(op._doubt) && applied.length > nApplied) markDoubt(op, applied[applied.length - 1])
    }

    // 〔@ref〕換成〔實際 id〕;所指之新項被終審剔除、被拒收或於同差量稍後才失敗者無 id 可換,去掉該標記(不留短命識別碼)。
    //   長度與同文比對以模型原文為準(上方逐操作已查),此處只改存入之文字。
    const linkRefs = (v) => (typeof v === 'string'
        ? v.replace(TEMP_REF_RE, (m, ref) => {
            const id = refState.get(ref)?.id
            return id ? `〔${id}〕` : ''
        })
        : v)
    // 理由類欄位(理由、沿革說明、存疑說明)中殘留之本批代號(「N4 之新證據」)換成筆記連結 [[id]]:代號只在本批有效,下一批同名之代號即指向別篇。
    //   內容欄位不換(檢定標準 ES17:知識內容不提及筆記——殘留代號由納入閘交修訂、定稿仍殘留即不採;不自動轉成連結)
    const linkCodes = (v) => linkRefs(typeof v === 'string' ? v.replace(/(?<![A-Za-z0-9])N\d+(?![0-9])/g, (m) => (codeMap.has(m) ? `[[${codeMap.get(m)}]]` : m)) : v)
    const linkFields = (obj, fields, fn = linkRefs) => {
        for (const f of fields) {
            if (isarr(obj?.[f])) obj[f] = obj[f].map(fn)
            else if (typeof obj?.[f] === 'string') obj[f] = fn(obj[f])
        }
    }
    for (const id of touched) {
        const hit = findItem(s, id)
        if (!hit) continue
        linkFields(hit.item, [...(TEXT_FIELDS[hit.key] || []), 'validPeriod'])
        linkFields(hit.item, ['statusNote'], linkCodes)
        for (const sd of isarr(hit.item.sides) ? hit.item.sides : []) linkFields(sd, ['position', 'conditions'])
        for (const h of isarr(hit.item.history) ? hit.item.history : []) if (h.version === version) linkFields(h, ['note'], linkCodes)
        if (isobj(hit.item.doubt) && hit.item.doubt.version === version) linkFields(hit.item.doubt, ['note'], linkCodes)
    }
    if (essenceSeen) linkFields(s.essence, ['text'])

    deriveStatus(s)
    // 只有實際改動才升版:confirm 之出處皆已在者為 noop(重審重新呈現已消化筆記時之常態),全為 noop 不升版、不寫沿革
    if (applied.some((a) => !a.noop)) {
        s.version = version
        s.updatedAt = at
        const byOp = {}
        for (const a of applied) byOp[a.op] = (byOp[a.op] || 0) + 1
        // 沿革記實際棒次(ctx.step;未給即套用機制);整併之水位只由 ctx.consolidate 真值寫——本質專步以整併機制套用但不是整併(終審 §3 D5h)
        s.changelog.push({ version, at, mode: str(ctx.step) || mode, batch: [...new Set(codeMap.values())], applied: byOp, rejected: rejected.length, withheld: withheld.length, ...(isobj(ctx.meta) ? ctx.meta : {}) })
        if (ctx.consolidate === true) s.lastConsolidated = version
    }
    // 暫緩之操作:@ref 換成本差量實際配發之 id(供存入待審;所依之新項被拒者 resolved 為 null;所依之新項同為暫緩者保留 @ref——同入待審、裁決時同一差量內可解析)
    const withheldRefs = new Set(withheld.map((w) => str(w._op?.ref)).filter(Boolean))
    for (const w of withheld) {
        const o = { ...w._op }
        delete o._i
        delete o._withhold
        delete o._withholdReason
        delete o._withholdEs9
        let ok = true
        const fix = (v) => {
            const x = str(v)
            if (!x.startsWith('@')) return x
            const id = refState.get(x.slice(1))?.id
            if (id) return id
            if (withheldRefs.has(x.slice(1))) return x
            ok = false
            return ''
        }
        for (const k of REF_FIELDS.ids) if (o[k] !== undefined) o[k] = fix(o[k])
        for (const k of REF_FIELDS.lists) if (isarr(o[k])) o[k] = o[k].map(fix)
        // 爭議各方之主張引用與出處:主張同上;出處由本批代號換成筆記 id(待審於日後另一批送審,代號不再對應)
        for (const key of ['sides', 'sides_add']) {
            if (!isarr(o[key])) continue
            o[key] = o[key].map((sd) => (isobj(sd) ? { ...sd, ...(isarr(sd.claims) ? { claims: sd.claims.map(fix) } : {}), ...(sd.sources !== undefined ? { sources: srcOf(sd.sources) } : {}) } : sd))
        }
        linkFields(o, ['reason'], linkCodes) // 待審於日後另一批送審:理由中之本批代號換成連結;內容欄只換〔@ref〕(ES17 同上)
        linkFields(o, ['validPeriod', 'note', 'text'])
        o.sources = w.sources
        w.resolved = ok ? o : null
        delete w._op
    }
    // 回報依差量原序號排列(處理順序為相序,與差量順序不同)
    const byIndex = (a, b) => a.index - b.index
    return { state: s, applied: applied.sort(byIndex), rejected: rejected.sort(byIndex), withheld: withheld.sort(byIndex), touched: [...touched], overLimit, lengthChecked }
}

/**
 * 暫緩之操作存入待審(終審漏裁決之終態操作;終審失敗時之破壞性操作——終態操作、大幅改寫之 revise、改為 resolved 之 dispute_update,
 *   見 standards/opGuards.opClass;其筆記因此算涵蓋——操作被保存,不是被丟棄)
 *
 * 同一目標(新增類操作以 ref)已有待審者,新者取代舊者——保留原登錄時刻(久懸可見)、tries＋1(每次再被暫緩,不論理由)、
 *   覆核判不了者 es9Tries＋1(w.es9);暫緩之理由取各操作自報者(w.why),無者取 opt.reason。超過上限丟最舊。
 *
 * @param {Object} state 輸入狀態(不修改,回傳新狀態)
 * @param {Array} withheld 輸入 applyDelta 之 withheld(各帶 resolved;why、es9 選填)
 * @param {Object} [opt={}] 輸入設定物件
 * @param {String} [opt.at=''] 輸入時刻
 * @param {String} [opt.reason=''] 輸入暫緩原因(如「終審失敗」;操作未自報理由時用)
 * @param {Array} [opt.prev] 輸入前一版之待審項(待審裁決重存時狀態之待審已清空,由此取原登錄時刻與次數;未給取狀態現有之待審)
 * @returns {Object} 回傳 { state, queued:Integer, dropped:Array(被丟之待審) }
 * @example
 * need test in nodejs.
 *
 * const next = queuePendingReview(state, withheld, { at: '2026-09-01T00:00:00+08:00', reason: '終審漏裁決' })
 * console.log(next.pendingReview.length)
 */
export function queuePendingReview(state, withheld, opt = {}) {
    if (!isobj(opt)) opt = {}
    const s = clone(state)
    const q = isarr(s.pendingReview) ? s.pendingReview : []
    const pool = isarr(opt.prev) ? opt.prev : q
    let queued = 0
    for (const w of isarr(withheld) ? withheld : []) {
        if (!w?.resolved) continue
        const key = pendingKeyOf(w.resolved)
        const k = q.findIndex((x) => pendingKeyOf(x.op) === key)
        const prev = k >= 0 ? q[k] : (pool.find((x) => pendingKeyOf(x?.op) === key) || null)
        const entry = {
            op: w.resolved, sources: w.sources || [], at: prev ? str(prev.at) : str(opt.at), version: s.version, reason: str(w.why) || str(opt.reason),
            tries: (Number(prev?.tries) || 0) + (prev ? 1 : 0), es9Tries: (Number(prev?.es9Tries) || 0) + (w.es9 === true ? 1 : 0),
            ...(prev ? { lastAt: str(opt.at) } : {}),
        }
        if (k >= 0) q.splice(k, 1, entry)
        else q.push(entry)
        queued++
    }
    const dropped = q.length > PENDING_REVIEW_CAP ? q.splice(0, q.length - PENDING_REVIEW_CAP) : []
    s.pendingReview = q
    return { state: s, queued, dropped }
}

/**
 * 待審項之鍵(同一目標只存一項;新增類以 ref)
 *
 * @param {Object} op 輸入操作
 * @returns {String} 回傳鍵
 */
function pendingKeyOf(op) {
    return `${str(op?.op)}|${str(op?.id) || str(op?.into) || str(op?.ref)}`
}

/**
 * 本批筆記之涵蓋計算:被已套用操作引用(used)、模型列入 skipped(理由須在 SKIP_REASONS,可附說明——見 skipReasonOf;回傳之 reason 為正規化後者)、被終審剔除之操作所引用(dropped)皆算涵蓋;
 * 其餘(含只被程式拒收或暫緩之操作引用者)為未涵蓋——值得重試(多為引用問題)
 *
 * @param {Object} o 輸入物件
 * @param {Array} o.batch 輸入本批 [{ code, id }]
 * @param {Array} [o.applied=[]] 輸入 applyDelta 之 applied(各帶 sources)
 * @param {Array} [o.skipped=[]] 輸入模型之 skipped [{ note:'N3'|id, reason }]
 * @param {Array} [o.dropped=[]] 輸入 applyVerdicts 之 dropped(各帶 sources 與 reason)
 * @param {Array} [o.queued=[]] 輸入存入待審之暫緩操作(applyDelta 之 withheld 中 resolved 非 null 者,各帶 sources)——操作已保存,其筆記算涵蓋
 * @param {Array} [o.removed=[]] 輸入終審 fix 刪去之出處 [{ note, reason }](終審已判定,算涵蓋)
 * @param {Array} [o.recheckIds=[]] 輸入本批中之重審筆記 id:其略過理由另收 RECHECK_SKIP_REASONS
 * @returns {Object} 回傳 { used:Array(id), pending:Array(id), skipped:Array({note,reason}), dropped:Array({note,reason}), uncovered:Array(id), badSkipped:Array }
 * @example
 * const cov = coverageOf({ batch: [{ code: 'N1', id: 'n-a' }, { code: 'N2', id: 'n-b' }, { code: 'N3', id: 'n-c' }], applied: [{ sources: ['n-a'] }], skipped: [{ note: 'N2', reason: '離題' }] })
 * console.log(cov.used, cov.skipped, cov.uncovered)
 * // => [ 'n-a' ] [ { note: 'n-b', reason: '離題' } ] [ 'n-c' ]
 */
export function coverageOf(o) {
    const recheck = new Set(isarr(o?.recheckIds) ? o.recheckIds.map((x) => str(x)) : [])
    const batch = (isarr(o?.batch) ? o.batch : []).filter((b) => isobj(b) && str(b.id))
    const byCode = codeMapOf(batch)
    const toNote = (x) => noteOfCode(byCode, x)
    const used = new Set((isarr(o?.applied) ? o.applied : []).flatMap((a) => (isarr(a?.sources) ? a.sources : [])).filter((id) => byCode.has(id)))
    const skipped = []
    const badSkipped = []
    const seen = new Set(used)
    for (const k of isarr(o?.skipped) ? o.skipped : []) {
        const id = toNote(k?.note)
        const raw = str(k?.reason)
        const reason = id ? skipReasonOf(raw, recheck.has(id) ? [...SKIP_REASONS, ...RECHECK_SKIP_REASONS] : SKIP_REASONS) : ''
        if (!reason) {
            badSkipped.push({ note: str(k?.note), reason: raw })
            continue
        }
        if (seen.has(id)) continue
        seen.add(id)
        skipped.push({ note: id, reason })
    }
    const pending = []
    for (const q of isarr(o?.queued) ? o.queued : []) {
        for (const x of isarr(q?.sources) ? q.sources : []) {
            const id = toNote(x)
            if (!id || seen.has(id)) continue
            seen.add(id)
            pending.push(id)
        }
    }
    const dropped = []
    const addDrop = (x, reason) => {
        const id = toNote(x)
        if (!id || seen.has(id)) return
        seen.add(id)
        dropped.push({ note: id, reason })
    }
    for (const d of isarr(o?.dropped) ? o.dropped : []) {
        for (const x of isarr(d?.sources) ? d.sources : []) addDrop(x, `${ACCEPT_DROP_PREFIXES.dropped}${str(d.reason)}`)
    }
    for (const r of isarr(o?.removed) ? o.removed : []) addDrop(r?.note, `${ACCEPT_DROP_PREFIXES.removed}${str(r?.reason) || '未說明'}`)
    const uncovered = batch.map((b) => str(b.id)).filter((id) => !seen.has(id))
    return { used: [...used], pending, skipped, dropped, uncovered, badSkipped }
}

/**
 * 略過理由正規化:合法理由本身,或「合法理由＋分隔符＋說明」(取開頭之合法理由);其餘回 ''
 *   模型常在合法理由後附說明(「離題：…，非對立」「離題(…)」「品質不足:…」;不少是依【關聯脈絡】之指示說明非對立)——
 *   逐字比對會把它們當未涵蓋而重送。理由與說明之間須有分隔符,
 *   免得「離題目很遠」之類被誤認
 *
 * @param {String} reason 輸入模型給之理由
 * @param {Array} allowed 輸入合法理由清單
 * @returns {String} 回傳正規化後之合法理由,不合法回 ''
 */
function skipReasonOf(reason, allowed) {
    if (allowed.includes(reason)) return reason
    return allowed.find((r) => reason.startsWith(r) && /^[\s:：,，;；(（\-—、]/.test(reason.slice(r.length))) || ''
}

/**
 * 本批落帳:已涵蓋與「未涵蓋達上限」者計入 consumed,略過／剔除／逾限之理由記入 skipped(不改版號)
 *
 * @param {Object} state 輸入狀態(不修改,回傳新狀態)
 * @param {Object} cov 輸入 coverageOf 之產出
 * @param {Object} [opt={}] 輸入設定物件
 * @param {Array} [opt.exhausted=[]] 輸入未涵蓋且已達重試上限之筆記 id(記「未涵蓋逾限」)
 * @returns {Object} 回傳新狀態
 * @example
 * need test in nodejs.
 *
 * const cov = coverageOf({ batch, applied, skipped })
 * const next = commitBatch(state, cov, { exhausted: [] })
 * console.log(next.consumed)
 */
export function commitBatch(state, cov, opt = {}) {
    const s = clone(state)
    if (!isobj(opt)) opt = {}
    const exhausted = isarr(opt.exhausted) ? opt.exhausted.map(str).filter(Boolean) : []
    const notes = [...(cov?.used || []), ...(cov?.pending || []), ...(cov?.skipped || []).map((x) => x.note), ...(cov?.dropped || []).map((x) => x.note), ...exhausted]
    s.consumed = uniq([...s.consumed, ...notes])
    // ⑰已消化者不再是夥伴候選(不論經何途徑入批)
    if (isobj(s.partners)) for (const id of notes) delete s.partners[id]
    const reasons = [...(cov?.skipped || []), ...(cov?.dropped || []), ...exhausted.map((note) => ({ note, reason: EXHAUSTED_REASON }))]
    for (const r of reasons) s.skipped.push({ note: r.note, reason: r.reason, version: s.version })
    return s
}
