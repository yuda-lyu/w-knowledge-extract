    integrate: { params: '(input, ctx)' },
    audit: { params: '(input, ctx)' },
    revise: { params: '(input, ctx)' },
    acceptAppendix: { params: '(input, ctx)', returns: '字串（無附錄時回 \'\'）' },
    supportCheck: { params: '(input, ctx)' },
    supportRevise: { params: '(input, ctx)' },
    orphanCheck: { params: '(input, ctx)' },
    disputeCheck: { params: '(input, ctx)' },
    synonymPropose: { params: '(names, ctx)' },
    synonymCheck: { params: '(items, ctx)' },
}).map(([k, v]) => [k, Object.freeze(v)])))

/** 提示詞覆寫之工廠名(錯誤訊息用) */
const OWNER = 'createExtractDomain'

/**
 * 一條規則之形狀問題(合格回 '')
 *
 * @param {*} r 輸入規則
 * @returns {String} 回傳問題說明
 */
function ruleProblemOf(r) {
    if (!isobj(r)) return '須為物件 { id, priority, text }'
    if (!isestr(r.id) || !r.id.trim()) return 'id 須為非空字串'
    if (!(typeof r.priority === 'number' && Number.isFinite(r.priority))) return `「${r.id}」之 priority 須為數字`
    if (typeof r.text !== 'string' || !r.text.trim()) return `「${r.id}」之 text 須為非空字串`
    return ''
}

/**
 * 規則表之形狀問題(合格回 ''):非空陣列、逐條合 ruleProblemOf、id 不重複
 *
 * @param {*} rules 輸入規則表
 * @returns {String} 回傳問題說明
 */
function rulesProblemOf(rules) {
    if (!isarr(rules) || !rules.length) return '須為非空陣列'
    const seen = new Set()
    for (const [i, r] of rules.entries()) {
        const p = ruleProblemOf(r)
        if (p) return `第 ${i + 1} 條：${p}`
        if (seen.has(r.id)) return `id「${r.id}」重複`
        seen.add(r.id)
    }
    return ''
}

/** 資料成員之形狀問題(成員 → (值) => 問題說明,合格回 '') */
const DATA_SHAPES = Object.freeze({
    kb: (v) => (isestr(v) && v.trim() ? '' : '須為非空字串（知識庫之稱呼）'),
    claimKinds: (v) => (isarr(v) && v.length === DEFAULT_CLAIM_KINDS.length && v.every((k, i) => k === DEFAULT_CLAIM_KINDS[i]) ? '' : `須為內建之主張種類（${DEFAULT_CLAIM_KINDS.join('、')}）——不可覆寫：規則表、檢定標準與核心版型皆以之為準`),
    limits: (v) => (isobj(v) && Object.keys(v).length && Object.values(v).every((x) => typeof x === 'number' && Number.isFinite(x) && x > 0) ? '' : '須為物件且各值為正數（長度上限）'),
    rules: (v) => {
        const p = rulesProblemOf(v)
        return p ? `須為規則陣列 [{ id, priority, text }]：${p}` : ''
    },
    evidence: (v) => (isobj(v) && isfun(v.apply) && isarr(v.levels) && v.levels.length && isfun(v.selfLimitOf) ? '' : '須為證據判定物件（具 apply、levels、selfLimitOf）'),
    weakEvidence: (v) => (typeof v === 'string' ? '' : '須為字串'),
    evidenceRule: (v) => {
        if (v === null) return ''
        try {
            if (isobj(v)) {
                noTopEvidenceOf({ caveats: ['—'] }, v)
                return ''
            }
        }
        catch { /* 不合規格之規則:同下 */ }
        return '須為摘要套件之證據規則（resolveExtractVocab 之 evidenceRule）或 null'
    },
    contentKeyVersion: (v) => (typeof v === 'string' ? '' : '須為字串（內容鍵之版本）'),
})

/**
 * 提煉 domain 之形狀問題(注入整組 domain 之檢核;成員表 EXTRACT_DOMAIN_MEMBERS):恆必填之成員缺之、函數成員非函數(選填者給了亦然)、
 *   資料成員之形狀不合(kb、claimKinds〔不可覆寫〕、limits、rules、evidence、weakEvidence、evidenceRule、contentKeyVersion)。
 *   依功能開啟而必填之成員與爭議覆核之「非對立之界線」只由功能之單一判定(extractFeatureProblems)判——不讀設定,同一規則不寫兩處
 *
 * @param {*} domain 輸入 domain
 * @returns {Array} 回傳問題說明陣列(無問題為空陣列)
 * @example
 * console.log(extractDomainShapeProblems(createExtractDomain({})))
 * // => []
 *
 * console.log(extractDomainShapeProblems({ ...createExtractDomain({}), claimKinds: ['rule'], kb: undefined }))
 * // => [ '缺成員：kb', 'claimKinds 須為內建之主張種類（principle、rule、pitfall、temporal）——不可覆寫：規則表、檢定標準與核心版型皆以之為準' ]
 */
export function extractDomainShapeProblems(domain) {
    if (!isobj(domain)) return ['domain 須為物件']
    const out = []
    const missing = Object.entries(EXTRACT_DOMAIN_MEMBERS).filter(([k, m]) => domain[k] === undefined && m.need === 'always').map(([k]) => k)
    if (missing.length) out.push(`缺成員：${missing.join('、')}`)
    for (const [k, m] of Object.entries(EXTRACT_DOMAIN_MEMBERS)) {
        if (domain[k] === undefined) continue
        if (m.kind === 'fn') {
            if (!isfun(domain[k])) out.push(`${k} 須為函數`)
            continue
        }
        const p = DATA_SHAPES[k](domain[k])
        if (p) out.push(`${k} ${p}`)
    }
    return out
}

/**
 * 覆寫之回傳檢查(呼叫期):可為空之字串(終審附錄、重審段——無內容時合法回 '';所需之回傳形式取 EXTRACT_PROMPT_SIGS 之 returns,訊息不另寫)
 *
 * @param {*} v 輸入覆寫函數之回傳值
 * @param {String} key 輸入鍵
 * @returns {String} 回傳同一字串
 */
function maybeEmptyTextOf(v, key) {
    if (typeof v !== 'string') throw new Error(`opt.prompts.${key} 須回傳${EXTRACT_PROMPT_SIGS[key].returns}`)
    return v
}

/**
 * 覆寫之回傳檢查(呼叫期):條件規則(規則物件或 null;回傳形式取 EXTRACT_PROMPT_SIGS 之 returns)
 *
 * @param {*} v 輸入覆寫函數之回傳值
 * @param {String} key 輸入鍵
 * @returns {Object|null} 回傳同一值
 */
function ruleOrNullOf(v, key) {
    if (v === null) return null
    const p = ruleProblemOf(v)
    if (p) throw new Error(`opt.prompts.${key} 須回傳${EXTRACT_PROMPT_SIGS[key].returns}（${p}）`)
    return v
}

/**
 * 覆寫之回傳檢查(建構期):規則表(opt.prompts.rules 之回傳;回傳形式取 EXTRACT_PROMPT_SIGS 之 returns)
 *
 * @param {*} v 輸入覆寫函數之回傳值
 * @returns {Array} 回傳同一陣列
 * @throws {Error} 不合規則陣列時拋出(附第一個問題)
 */
function rulesOf(v) {
    const p = rulesProblemOf(v)
    if (p) throw new Error(`opt.prompts.rules 須回傳${EXTRACT_PROMPT_SIGS.rules.returns}（${p}）`)
    return v
}


/**
 * 建立提煉之內建領域預設
 *
 * 回傳物件之全部成員皆可由組裝者整組置換(cfg.domains;成員與必填條件見 EXTRACT_DOMAIN_MEMBERS);提示詞與規則表可個別覆寫(opt.prompts)
 *
 * @param {Object} [opt={}] 輸入設定物件
 * @param {Object} [opt.vocab=null] 輸入詞彙表覆寫物件(見 resolveExtractVocab;他段共用之詞彙表先以 pickExtractVocab 取子集)，取其 kbLabel／domain 作為
 *   知識庫稱呼、guide.extract 作為提示詞之領域句、evidenceLevels／evidenceCaps／basisCaps／selfLimitPhrases 作為證據判定、presumedBasis 與
 *   evidenceRule 作為推定、absolutePhrases 作為絕對語氣指標，預設null代表全用內建
 * @param {Object} [opt.limits] 輸入長度上限覆寫(逐鍵)
 * @param {Object} [opt.prompts] 輸入提示詞覆寫(鍵與函數形式、回傳之檢查見 EXTRACT_PROMPT_SIGS):各為函數,收原參數＋末參數
 *   ctx { kbLabel, domain, vocab, defaultPrompt(內建函數,參數同原成員) };未給者用內建
 * @param {String|Object} [opt.locale='zh-TW'] 輸入語言設定(語言代碼或抓取套件 resolveLocale 之產物;覆核類提示詞之理由語言句取其稱呼)
 * @returns {Object} 回傳 domain 物件:資料成員 kb、claimKinds、limits、rules、evidence、weakEvidence、evidenceRule、contentKeyVersion;
 *   函數成員 noteDigest、renderState、toneOf、buildProposePrompt、buildIntegratePrompt、buildAuditPrompt、buildRevisePrompt、buildReviewPrompt、
 *   buildAcceptAppendix、buildConsolidatePrompt、buildEssencePrompt、buildRecheckText、withdrawnRule、buildSupportCheckPrompt、buildSupportRevisePrompt、
 *   buildOrphanCheckPrompt、buildDisputeCheckPrompt、buildSynonymProposePrompt、buildSynonymCheckPrompt、renderTwinRedirect(各成員之必填條件見 EXTRACT_DOMAIN_MEMBERS)
 * @throws {Error} opt.vocab 不合規格(見 resolveExtractVocab)、opt.prompts 不合規格(非物件、不認得之鍵、值非函數)、或 opt.prompts.rules 之回傳
 *   不合規則陣列時拋出;覆寫之提示詞函數於呼叫時回傳不合者亦拋(見 EXTRACT_PROMPT_SIGS)
 * @example
 * const d = createExtractDomain({})
 * console.log(d.claimKinds)
 * // => [ 'principle', 'rule', 'pitfall', 'temporal' ]
 *
 * const d2 = createExtractDomain({ prompts: { audit: (input, ctx) => `${ctx.defaultPrompt(input)}\n（附註）` } })
 * console.log(d2.buildAuditPrompt({ concept: '甲' }).endsWith('（附註）'))
 * // => true
 */
export function createExtractDomain(opt = {}) {

    //check
    if (!isobj(opt)) {
        opt = {}
    }

    const vocab = resolveExtractVocab(opt.vocab)
    const custom = promptOverridesOf(opt.prompts, EXTRACT_PROMPT_SIGS, OWNER)
    const kb = kbLabelOf(vocab)
    const L = resolveLocale(opt.locale, `${OWNER} 之 opt.locale`).label // 輸出語言之稱呼(預設「繁體中文」,提示詞逐字不變)
    const g = vocab.guide.extract // 提示詞之領域句(安裝方可經 vocab.guide.extract 逐欄覆寫;輸出格式留在本檔)
    const limits = { ...DEFAULT_LIMITS, ...(isobj(opt.limits) ? opt.limits : {}) }
    const claimKinds = [...DEFAULT_CLAIM_KINDS]
    /** 覆寫函數之末參數(抓取套件之 promptCtxOf;defaultPrompt＝該成員之內建函數) */
    const ctxOf = (defaultPrompt) => promptCtxOf(vocab, kb, defaultPrompt)
    // 規則表:內建或覆寫(opt.prompts.rules;建構時套用一次並驗形狀)——各步之提示詞同讀此一份
    const builtinRules = () => defaultRules(vocab, limits)
    const rules = custom.rules ? rulesOf(custom.rules(builtinRules(), ctxOf(builtinRules))) : builtinRules()
    // 爭議覆核(ES14)之判準:生效之規則表之「非對立之界線」(呼叫端可另傳 boundary)
    const defaultBoundary = (rules.find((r) => r.id === 'disputeBoundary') || {}).text || ''
    const levels = isarr(vocab.evidenceLevels) ? vocab.evidenceLevels : []
    const lowest = levels[levels.length - 1]
    const scopeWord = (scope) => (scope === 'category' ? '類別' : '概念')
    const absolutes = (isarr(vocab.absolutePhrases) ? vocab.absolutePhrases : []).map(String).filter(Boolean)

    // 推論類之證據性質(vocab.inferenceBasis;出處支持之覆核 ES9／ES8 以此分判準,見 buildSupportCheckPrompt)
    const inferWords = (isarr(vocab.inferenceBasis) ? vocab.inferenceBasis : []).filter(Boolean)
    const inferenceOf = (basis) => inferWords.some((k) => String(basis || '').includes(k))
    // 受檢項之證據性質標示(判準依類別;類別由程式判,不交模型判)
    const basisNote = (basis) => `證據性質：${String(basis || '').trim() || '未標'}；${inferenceOf(basis) ? '推論類' : '非推論類'}`
    // 「受支持」之判準(ES9 出處支持、ES8 失撐找支持同一判準;依 basis 類別)。同義改寫與擴大範圍之界線須明文:未明文時,用詞較概括之同義改寫
    //   會被判語氣超出、寫了推算前提之非推論類會被放行
    const supportRule = (src) => [
        `  ・非推論類之項：${src}本身寫出了斷言的結論（同義改寫亦可）。斷言即使寫了前提或推算過程，只要結論須自行計算或推論才得到，就不算（理由寫明「須推論」）。`,
        ...(inferWords.length ? [`  ・推論類之項：斷言可由${src}明載的內容推得，且斷言本身寫明了推論所依的前提（前提須為${src}明載）；只寫結論而未寫前提者不算受支持（理由寫明「未寫前提」）。`] : []),
        `  ・兩類皆須：斷言所附的條件、期間與${src}相符。`,
        `  ・同義改寫＝措辭不同而所指之對象、範圍、條件與${src}相同；以概括之名稱指稱${src}所測之分組或樣本（範圍未擴大），仍是同義改寫。`,
    ].join('\n')

    // 語氣標記(見檔頭):md 渲染使用
    const basisKeys = Object.keys(isobj(vocab.basisCaps) ? vocab.basisCaps : {}).filter((k) => k !== '*')
    const natureTags = (texts) => {
        const t = texts.filter((x) => typeof x === 'string' && x).join('\n')
        return basisKeys.filter((k) => t.includes(k)).map((k) => `〔${k}〕`).join('')
    }
    const typesOf = (ids, notesById) => (isarr(ids) ? ids : []).map((id) => notesById.get(id)?.claimType).filter(Boolean)
    // 推定(見檔頭【語氣標記】):出處筆記(現存者)皆自承無最高證據等級之要件 → 標推定詞(vocab.presumedBasis)。逐篇之判定由呼叫端以
    //   ctx.noTopOf 給(摘要套件之 noTopEvidenceOf——與摘要之最高級要件同一判準;材料為筆記 md 之 caveats 與證據說明,讀檔與快取屬機制)。
    //   只推保守方向:自承缺最高級之要件者,其證據性質不可能是最高級;推定詞須與要件成對,由安裝方指定,未指定即不推定
    const presumed = vocab.presumedBasis
    const presumedOf = (ids, notesById, noTopOf) => {
        if (!presumed) return false
        const ns = (isarr(ids) ? ids : []).filter((id) => notesById.has(id))
        return ns.length > 0 && ns.every((id) => noTopOf(id) === true)
    }
    const weakOf = (x) => ((x?.evidence?.level || UNASSESSED) === lowest ? '有一說：' : '')
    const claimMark = (c, notesById, noTopOf) => natureTags([String(c?.basis || '')]) || natureTags(typesOf(c?.sources, notesById)) || (presumedOf(c?.sources, notesById, noTopOf) ? natureTags([presumed]) : '')
    const sideMark = (sd, claimOf, notesById, noTopOf) => {
        const cited = (isarr(sd?.claims) ? sd.claims : []).map((id) => claimOf.get(id)).filter(Boolean)
        const own = natureTags([String(sd?.basis || ''), ...typesOf(sd?.sources, notesById)]) || (presumedOf(sd?.sources, notesById, noTopOf) ? natureTags([presumed]) : '')
        return natureTags([own, ...cited.map((c) => claimMark(c, notesById, noTopOf))])
    }

    /**
     * 絕對語氣指標(只計數不擋——否定句會誤判,故不做硬擋):文字中第一個非否定之絕對語氣片語
     * 片語之前兩字含「不未非無」或為「難以」者視為否定(如「不一定會」「並非總是」「難以保證會」)
     *
     * @param {String} text 輸入文字
     * @returns {String} 回傳命中之片語，無則空字串
     */
    function toneOf(text) {
        const t = String(text || '')
        for (const p of absolutes) {
            for (let i = t.indexOf(p); i >= 0; i = t.indexOf(p, i + 1)) {
                const prev = t.slice(Math.max(0, i - 2), i)
                if (!/[不未非無]/.test(prev) && prev !== '難以') return p
            }
        }
        return ''
    }

    /**
     * 筆記分節摘要:依章名取各節並限長(衝突章不會因位於文末而被截掉);列出 frontmatter conflicts;
     * 找不到任何已知章節(自訂筆記版型)→ 退回截斷正文(sectioned:false,由機制記 WARN)
     *
     * @param {Object} note 輸入筆記記錄
     * @param {Object|null} md 輸入 readMd 之產物(讀不到為 null)
     * @param {Object} [ctx={}] 輸入 { code:'N1', codeOf:(id)=>代號或'', titleOf:(id)=>標題 }
     * @returns {Object} 回傳 { text, sectioned, contentKey }:contentKey＝起草者讀得到之「筆記內容」(各章截斷後文字＋樣本期、品質疑慮、發布;
     *   不含標題、內容類型、證據等級——重標所改,走投影更新——亦不含衝突章與 frontmatter conflicts——關聯段之投影),換行與空白正規化;
     *   md 讀不到為 null(讀不到不等於內容被清空,不可當成內容變更)
     */
    function noteDigest(note, md, ctx = {}) {
        const head = [
            `### ${ctx.code || ''} ${note?.title || ''}`,
            // 筆記之缺值以摘要之缺值字樣(摘要套件之 NOTE_PLACEHOLDERS;與筆記落地版型同一字樣)
            `標注：${note?.claimType || NOTE_PLACEHOLDERS.claimType}｜證據 ${note?.evidenceLevel || NOTE_PLACEHOLDERS.evidenceLevel}｜樣本期 ${note?.samplePeriod || NOTE_PLACEHOLDERS.samplePeriod}${(note?.caveats || []).length ? `｜⚠ ${note.caveats.join('/')}` : ''}`,
            `來源：${note?.sourceName || ''}｜發布：${md?.front?.published || note?.published || '（未提供）'}`,
        ]
        // 換行先統一為 LF 再擷取與截斷(CRLF 之 \r 會佔截斷字數、改變截斷點;checkout 之換行格式不可改變起草者所見)
        const body = md ? dropSection(String(md.body ?? '').replace(/\r\n?/g, '\n'), RELATE_SECTION_TITLES.relations) : ''
        const keyHead = keyNorm(`樣本期 ${note?.samplePeriod || ''}｜${(note?.caveats || []).join('/')}｜發布 ${md?.front?.published || note?.published || ''}`)
        const parts = []
        const keyParts = []
        for (const [title, max] of NOTE_SECTIONS) {
            const sec = sectionOf(body, title)
            if (!sec) continue
            const text = sec.text.replace(/^## [^\n]*\n/, '').trim()
            if (!text) continue
            parts.push(`【${title}】\n${strTruncate(text, max)}`)
            if (title !== PROJECTED_SECTION) keyParts.push(`【${title}】${keyNorm(strTruncate(text, max))}`)
        }
        const conflicts = isarr(md?.front?.conflicts) ? md.front.conflicts.map(String) : []
        if (conflicts.length) {
            const who = conflicts.map((id) => {
                const c = typeof ctx.codeOf === 'function' ? ctx.codeOf(id) : ''
                return c ? `${c}（同批）` : `〈${(typeof ctx.titleOf === 'function' && ctx.titleOf(id)) || id}〉（非本批）`
            })
            parts.push(`衝突對象：${who.join('、')}`)
        }
        if (!parts.length || (parts.length === 1 && conflicts.length)) {
            // 退回路徑(無已知章節之自訂版型):內容鍵取去掉兩個投影章(關聯、衝突章)後之正文,同截斷
            const fallbackKey = md ? keyNorm(strTruncate(dropSection(body, PROJECTED_SECTION).trim(), 1800)) : null
            return { text: [...head, strTruncate(body.trim(), 1800)].join('\n'), sectioned: false, contentKey: md ? `${keyHead}｜${fallbackKey}` : null }
        }
        return { text: [...head, ...parts].join('\n'), sectioned: true, contentKey: md ? `${keyHead}｜${keyParts.join('')}` : null }
    }

    /**
     * 重審段(只在本批有重審筆記時附於起草、審計、修訂、終審提示詞之末;無則不出現——一般批提示詞逐字不變)
     *
     * @param {Array} items 輸入 [{ code, reason:'changed'|'conflict'|'conflictGone'|'resend'|'relationChanged'|'conflictReason'|'partner', citedBy:[條目 id], anchorTitle,
     *   anchorItems:[條目 id], kind?, newType?, why? }]
     * @param {Object} [opt={}] 輸入 { stage:'draft'|'review' }
     * @returns {String} 回傳段落文字(無項回'')
     */
    function buildRecheckText(items, opt = {}) {
        const list = isarr(items) ? items.filter((x) => isobj(x) && x.code) : []
        if (!list.length) return ''
        const ref = (ids) => (isarr(ids) && ids.length ? `〔${ids.join('〕〔')}〕` : '（無）')
        const relWord = (x) => (x.kind === 'retire' ? '已由關聯段撤銷' : x.kind === 'retype' ? `已改為「${x.newType}」` : '理由已更新')
        const line = (x) => (x.reason === RECHECK_REASONS.changed
            ? `- ${x.code}（內容已更新）：本核心引用它之條目 ${ref(x.citedBy)}。`
            : x.reason === RECHECK_REASONS.conflictGone
                ? `- ${x.code}（衝突已撤銷）：與核心已引之〈${x.anchorTitle || ''}〉之衝突關係已由關聯段重評${x.kind === 'retire' ? '撤銷' : `改為「${x.newType}」`}${x.why ? `（理由：${x.why}）` : ''}；涉及之條目 ${ref(x.anchorItems)}。`
                : x.reason === RECHECK_REASONS.resend
                    ? `- ${x.code}（重送）：前次提煉多次未交代此篇之去向而出隊，今重送一次。`
                    : x.reason === RECHECK_REASONS.relationChanged
                        ? `- ${x.code}（關聯已變更）：此篇當初經與核心已引之〈${x.anchorTitle || ''}〉之關聯入批，該關聯${relWord(x)}${x.why ? `（${x.kind === 'retire' ? '撤銷理由' : '現理由'}：${x.why}）` : ''}；本核心引用它之條目 ${ref(x.citedBy)}。`
                        : x.reason === RECHECK_REASONS.conflictReason
                            ? `- ${x.code}（衝突理由已更新）：與核心已引之〈${x.anchorTitle || ''}〉之衝突關係理由已更新${x.why ? `為：${x.why}` : ''}；涉及之條目 ${ref(x.anchorItems)}。`
                            : x.reason === RECHECK_REASONS.partner
                                ? `- ${x.code}（新關聯）：此篇未曾入本核心，因與核心已引之〈${x.anchorTitle || ''}〉（${ref(x.anchorItems)}之出處）新判「${x.newType}」關係而呈現${x.why ? `（理由：${x.why}）` : ''}。`
                                : `- ${x.code}（新判衝突）：與核心已引之〈${x.anchorTitle || ''}〉（${ref(x.anchorItems)}之出處）於【關聯脈絡】判為衝突。`)
        // 衝突已撤銷、重送、關聯變更之說明只在本批有此型時附上(無此型之批,其重審段不隨之而變;檢定標準 ES15、ES10)
        const gone = list.some((x) => x.reason === RECHECK_REASONS.conflictGone)
        const resend = list.some((x) => x.reason === RECHECK_REASONS.resend)
        const relChanged = list.some((x) => x.reason === RECHECK_REASONS.relationChanged)
        const reasonUpd = list.some((x) => x.reason === RECHECK_REASONS.conflictReason)
        const partner = list.some((x) => x.reason === RECHECK_REASONS.partner)
        // 只有內容更新與新判衝突時,標頭與各行為該兩型之固定字樣;有關聯變更之理由(ES10)時,標頭依實況(新關聯者未曾入本核心)、
        //   各型之說明只列本批有者
        const anyNew = relChanged || reasonUpd || partner
        const onlyPartner = list.every((x) => x.reason === RECHECK_REASONS.partner)
        const has = (r) => list.some((x) => x.reason === r)
        const why = `因內容更新${relChanged || reasonUpd ? '、關聯變更' : ''}或新判衝突`
        if (opt?.stage === 'review') {
            const head = !anyNew ? '（本批有被本核心消化過、因內容更新或新判衝突而重新呈現之筆記）'
                : `（本批有${partner ? '因與核心已引之筆記新判關聯而呈現之新筆記' : ''}${partner && !onlyPartner ? '，及' : ''}${onlyPartner ? '' : `被本核心消化過、${why}而重新呈現之筆記`}）`
            const base = onlyPartner ? '' : '審核重審筆記相關之操作時：依其新內容判斷 revise／retract 是否成立；仍有他篇出處之條目不得以「無出處支持」撤回；新判衝突者應立爭議或明示非對立。'
            const tail = [
                base,
                gone ? '衝突已撤銷者：不得依已撤之衝突新立爭議；拆解爭議須寫明兩方為何已不再對立。' : '',
                resend ? '重送者：須有去向（被引為出處，或列入 skipped 並寫理由）。' : '',
                relChanged ? '關聯已變更者：以「離題」撤回須確認該篇內容確與本核心主題無關（不得只因關聯撤銷即撤回）。' : '',
                reasonUpd ? '衝突理由已更新者：拆解或改寫爭議須依新理由寫明兩方之關係。' : '',
                partner ? '新關聯者：被引為出處之內容須確實出自該篇；與本核心主題無關者應列入 skipped（離題），不得硬併入條目。' : '',
            ].join('')
            return `\n\n${[`【重審】${head}`, ...list.map(line), tail].join('\n')}`
        }
        const head = !anyNew ? '（以下筆記已被本核心消化過，因內容更新或新判衝突而重新呈現；其餘筆記照一般規則處理）'
            : onlyPartner ? '（以下筆記未曾入本核心，因與核心已引之筆記新判關聯而呈現；本批其餘筆記照一般規則處理）'
                : `（以下筆記${partner ? '中，標「新關聯」者未曾入本核心、因與核心已引之筆記新判關聯而呈現，其他' : ''}已被本核心消化過，${why}而重新呈現；本批其餘筆記照一般規則處理）`
        const parts = [
            `【重審】${head}`,
            ...list.map(line),
            !anyNew || has(RECHECK_REASONS.changed) ? '- 內容已更新者：逐條核對所列條目——仍成立者不動或 confirm；與新內容不符者 revise；僅以該篇為出處且已不成立者 retract（理由「無出處支持」）；仍有他篇出處者不可撤回、只可 revise；新內容中之新知識照常 add。' : '',
            !anyNew || has(RECHECK_REASONS.conflict) ? '- 新判衝突者：依爭議規則立爭議（該篇之立場以該篇代號為出處），或判為非對立。' : '',
            resend ? '- 重送者：照一般規則處理，務必交代去向——被引為出處（新增、確認或修正條目），或列入 skipped 並寫理由。' : '',
            gone ? `- 衝突已撤銷者：核對依此衝突所立之爭議是否仍成立——兩方確已不再對立者以 dispute_dissolve 拆解（reason 寫明兩方各回答什麼問題、或為何可同時成立），並於 skipped 寫「${RECHECK_SKIP.notOpposed}」；仍對立者維持，並於 skipped 寫「${RECHECK_SKIP.unchanged}」。` : '',
            relChanged ? `- 關聯已變更者：核對此篇是否仍屬本核心之主題——其內容與本核心相關者，所列條目仍成立即不動，並於 skipped 寫「${RECHECK_SKIP.unchanged}」；其內容與本核心無關（當初只因該關聯入批）者，僅以該篇為出處之條目 retract（理由「離題」），仍有他篇出處者只可 revise 去除只出自該篇之內容。` : '',
            reasonUpd ? `- 衝突理由已更新者：依新理由核對所涉爭議之問題與兩方立場——仍準確者不動，並於 skipped 寫「${RECHECK_SKIP.unchanged}」；須補充說明者以 dispute_update 寫 note（附該篇代號為出處）；兩方已不再對立者以 dispute_dissolve 拆解並於 skipped 寫「${RECHECK_SKIP.notOpposed}」。` : '',
            partner ? '- 新關聯者：照一般規則處理——內容與本核心既有條目相關者引為出處（confirm 補出處、revise 補條件或限制、或 add 新條目）；與本核心主題無關者列入 skipped（理由「離題」）。' : '',
            onlyPartner ? '' : `- 重審筆記之去向另可列入 skipped 之理由：${RECHECK_SKIP_REASONS.join('｜')}（前者＝內容已更新而所列條目皆仍成立；後者＝新判衝突而兩方並非對立）。${partner ? '新關聯者為新筆記，不適用此二理由。' : ''}`,
        ].filter(Boolean)
        return `\n\n${parts.join('\n')}`
    }

    /**
     * 條件規則:核心有「出處已撤」之條目時加入規則表(優先序排最後,不改既有規則之編號;無此類條目時不出現)
     *
     * @returns {Object} 回傳規則 { id, priority, text }
     */
    const withdrawnRule = () => ({ id: 'withdrawn', priority: 99, text: '標「出處已撤」之條目（其出處筆記已被刪除）：本批筆記支持者以 confirm 補出處；本批無相關者不動；確定不成立者 retract（理由「無出處支持」）。出處已撤之條目不得被合併，也不得被新本質、新爭議方引用。' })

    /**
     * 起草提示詞:目前狀態＋本批筆記 → 差量(3+3 之三家獨立起草共用同一份)
     *
     * @param {Object} ctx 輸入 { concept, scope, digest(stateDigest 之文字), batch:[{ code, digest }], rulesText, relations(關聯脈絡文字,選填) }
     * @returns {String} 回傳 prompt 字串
     */
    function buildProposePrompt(ctx) {
        const batch = isarr(ctx?.batch) ? ctx.batch : []
        // 關聯脈絡:本批筆記之間、以及與核心已引筆記之間之關聯;衝突邊須呈現為爭議或說明非對立
        const rel = String(ctx?.relations || '').trim()
        // 檢定標準 ES5:兩端皆可引之衝突列冠 R 編號(關聯脈絡之列,workflow/relationContext),須逐條於 conflicts 交代去向;無 R 列時不附衝突之去向段
        const hasR = /^- R\d+ /m.test(rel)
        return `你是${kb}的提煉器。以下是「${ctx?.concept}」這個${scopeWord(ctx?.scope)}目前之核心知識狀態，以及本批 ${batch.length} 篇尚未提煉之筆記（代號 N1～N${batch.length}）。請只輸出「差量操作」，把本批筆記之知識併入核心；程式會逐條套用並檢查，不合規者拒收。

【規則】（依優先序；衝突時以前者為準）
${ctx?.rulesText || ''}

${proposeOpsText(claimKinds)}

【本批筆記之去向】每一篇都要交代：被某操作引為出處（支持既有條目者一律 confirm），或列入 skipped 並寫理由（${SKIP_REASONS.join('｜')}）。${hasR ? `

【衝突之去向】【關聯脈絡】中標 R 編號之衝突（兩端皆可引），每一條都要交代去向並列入 conflicts：${dispositionsText}。程式依套用結果核對，未處置者下次再呈現。` : ''}

【輸出格式】只回覆一個 JSON 物件，不要任何其他文字：
{"ops":[…],"skipped":[{"note":"N3","reason":"${SKIP_REASONS[0]}"}]${hasR ? ',"conflicts":[{"edge":"R1","disposition":"立爭議","reason":"…"}]' : ''}}

【目前核心狀態】
${ctx?.digest || '（尚無內容）'}

【本批筆記】
${batch.map((b) => b.digest).join('\n\n')}${rel ? `

【關聯脈絡】（關聯段判定之筆記間關係；標「衝突」者須依爭議規則處理——立爭議，或於 skipped／reason 說明為何非對立）
${rel}` : ''}`
    }

    /**
     * 終審提示詞:逐操作裁決(keep／drop／doubt／fix);不得新增操作、fix 不得增加出處;剔除只用於內容不成立者,標籤錯誤以 fix 改正
     *
     * @param {Object} ctx 輸入 { concept, scope, mode:'delta'|'pending'|'consolidate', ops:Array, batch:[{ code, digest }], touched(被觸及之既有條目摘要), rulesText }
     * @returns {String} 回傳 prompt 字串
     */
    function buildReviewPrompt(ctx) {
        const ops = isarr(ctx?.ops) ? ctx.ops : []
        const batch = isarr(ctx?.batch) ? ctx.batch : []
        const mode = ctx?.mode || 'delta'
        // fix 可改之欄位:只列本批出現之操作,取自程式之 FIXABLE_FIELDS(提示詞與程式同一來源)
        const fixable = [...new Set(ops.map((o) => o?.op))].filter((k) => FIXABLE_FIELDS[k]).map((k) => `${k}＝${FIXABLE_FIELDS[k].join('、')}`)
        const kindEx = claimKinds[Math.min(2, claimKinds.length - 1)]
        const lead = {
            pending: `以下是「${ctx?.concept}」前次未經終審裁決而暫緩之取代／撤回操作（逐條編號 i），以及其所引之筆記與被觸及之既有條目。`,
            consolidate: `以下是整併員對「${ctx?.concept}」提出之整併操作（逐條編號 i）與被觸及之既有條目；整併不處理筆記、不新增知識——請確認被合併者確為同義或重複、精簡後未改原意。`,
            essence: `以下是整併員為「${ctx?.concept}」所寫之本質（逐條編號 i）與其所引之既有條目；本步只寫本質、不處理筆記、不新增知識——請確認本質綜合所引之主張、未擇一抹平爭議、未加入所引主張沒有之內容。`,
        }[mode] || `以下是提煉器對「${ctx?.concept}」提出之差量操作（逐條編號 i），以及本批筆記與被觸及之既有條目。`

- keep：保留；
- drop：剔除——只用於內容不成立者，reason 擇一：${DROP_REASONS.join('｜')}（「性質標錯」指操作類別用錯，如應立爭議卻寫成取代或待解問題；新增主張 add、參數 param_add 只能以「${NEW_ITEM_DROP_REASONS.join('」「')}」剔除——其種類、證據性質、快照標錯者請以 fix 改正，不可剔除；爭議類操作 dispute_add、dispute_update、contest 只能以「${DISPUTE_DROP_REASONS.join('」或「')}」剔除（須有嚴重度「高」之離題意見）——不可把真實之對立藏掉；拆解 dispute_dissolve 若對立其實成立，以「對立成立」剔除）；
- doubt：只用於爭議類操作——reason 擇一：「${DOUBT_REASONS[0]}」（你依規則表「非對立之界線」認為兩方並非對立）｜「${DOUBT_REASONS[1]}」（兩方之對立只出自同一研究自身之正反，如同一篇之利弊、條件與限制），note 寫明兩方各回答什麼問題、或為何可同時成立、或出自同一研究之何處；程式照常套用該操作，並在爭議上標示終審存疑，不會移除（拆解另由提煉器提出）；

被其他操作以 "@ref名" 引用之新項若剔除，引用它之操作會一併不成立（連帶拒收）——內容可用而標籤或措辭有誤者，請以 fix 改正，不要剔除。
${TERMINAL_OPS.map((op) => `${TERMINAL_OP_LABELS[op]}（${op}）`).join('、')}不可逆，須逐條明列裁決；未列出者視為未審、本次不套用。${mode === 'pending' ? '其中材料仍不足以確定成立與否者（如出處載明尚待確認或核對），不要列出——留待審，待新材料再判，不可憑推測定案。' : ''}其餘操作未列出者視為 keep。不得新增操作，不要重寫整份核心。

【規則】（與提煉器相同；依優先序）
${ctx?.rulesText || ''}

【輸出格式】只回覆一個 JSON 物件，不要任何其他文字：
{"verdicts":[{"i":0,"action":"keep","reason":"","note":""}]}

【操作】
${ops.map((o, i) => `i=${i} ${JSON.stringify(o)}`).join('\n')}

【被觸及之既有條目】
${ctx?.touched || '（無）'}

【筆記】
${batch.map((b) => b.digest).join('\n\n') || '（無）'}`
    }

    /**
     * 整併提示詞:有效主張逾上限時合併重複、重排面向、精簡措辭;不處理筆記、不新增知識(3+3 之整併起草共用同一份)
     *
     * @param {Object} ctx 輸入 { concept, scope, digest(完整摘要), live(有效主張數), cap, rulesText }
     * @returns {String} 回傳 prompt 字串
     */
    function buildConsolidatePrompt(ctx) {
        return `你是${kb}的整併員。「${ctx?.concept}」這個${scopeWord(ctx?.scope)}之核心已累積 ${ctx?.live ?? '?'} 條有效主張（上限 ${ctx?.cap ?? '?'}），請整併以控制篇幅，不新增知識、不處理筆記：
- merge 合併同義或重複之主張／參數／待解問題：{"op":"merge","into":"C3","from":["C7","C9"],"reason":"…","text":"合併後文字（選填）"}（出處由程式取聯集；不可刪除有效主張之知識）
- refacet 面向改名或合併：{"op":"refacet","from":"舊面向","to":"新面向"}
- revise 精簡措辭（不改原意）：{"op":"revise","id":"C3","fields":{"text":"…"},"reason":"…"}
- essence 重寫本質（claims 只列主張，不列爭議）：{"op":"essence","text":"…","claims":["C1","C4"],"reason":"…"}
- related_prune 移除不相干之相關概念：{"op":"related_prune","concepts":["…"]}
- dispute_dissolve 拆解非對立之爭議（依規則表之爭議規則與「非對立之界線」判定；reason 寫明兩方各回答什麼問題、或為何可同時成立——標有「終審存疑」者請獨立判斷，不可只引述存疑；只有立場文字之方轉為主張，kinds[k]、annotations[k] 為其種類與標注，同新增主張之必填規則，引主張之方填 null）：{"op":"dispute_dissolve","id":"D2","reason":"…","kinds":[null,"${claimKinds[1] || claimKinds[0]}"],"annotations":[null,{"critique":"…","pros":["…"],"cons":["${ANNOTATION_UNSTATED}"]}]}（爭議轉入沿革；內容與出處全留）
爭議之各方不可合併成一方；有爭議之主張不可與其對立方合併；真實之對立不可拆解。

【規則】（依優先序）
${ctx?.rulesText || ''}

【輸出格式】只回覆一個 JSON 物件，不要任何其他文字：
{"ops":[…]}

【目前核心狀態（完整）】
${ctx?.digest || '（尚無內容）'}`
    }

    /**
     * 本質專步之提示詞(檢定標準 ES6:有效主張而無本質、或本質待重寫之核心——閒置核心不被選題選中,不另排則永不補):
     *   只寫本質、不處理筆記、不新增知識;給完整狀態(本質須綜合所引之主張)
     *
     * @param {Object} ctx 輸入 { concept, scope, digest(完整摘要), live(有效主張數), need:'missing'|'stale', why(待重寫之原因), rulesText }
     * @returns {String} 回傳 prompt 字串
     */
    function buildEssencePrompt(ctx) {
        const lead = ctx?.need === 'stale'

${sources}

【待覆核之爭議】
${items}`
    }

    /**
     * 自狀態渲染核心 md(純函數:時刻取自狀態,同一狀態與同一出處判定渲染逐字相同)
     *
     * @param {Object} state 輸入核心狀態
     * @param {Object} [ctx={}] 輸入 { notesById:Map(筆記 id → 記錄,供「提煉自」之標題與來源、語氣標記之內容類型),
     *   noTopOf:(筆記 id) => Boolean(該篇是否自承無最高證據等級之要件;推定之判準,見檔頭【語氣標記】——詞彙表給了推定詞時必給) }
     * @returns {Object} 回傳 { front, body }
     * @throws {Error} 詞彙表給了推定詞(presumedBasis)而 ctx.noTopOf 非函數時拋出(不可靜默不推定)
     */
    function renderState(state, ctx = {}) {
        const s = state || {}
        const notesById = ctx?.notesById instanceof Map ? ctx.notesById : new Map()
        if (presumed && typeof ctx?.noTopOf !== 'function') throw new Error(`renderState 需要 ctx.noTopOf(筆記 id) => Boolean：詞彙表之推定詞（presumedBasis「${presumed}」）已設，推定須知各出處筆記是否自承無最高證據等級之要件`)
        const noTopOf = typeof ctx?.noTopOf === 'function' ? ctx.noTopOf : () => false
        const TERM = new Set(TERMINAL_STATUSES)
        // 出處已撤:一律只列存活出處;失撐主張移至「出處已撤之條目」節、爭議方／參數／問題就地標示(皆為衍生,還原即回復)
        const W = withdrawnOf(s)
        const un = unsupportedOf(s)
        const alive = (ids) => (ids || []).filter((id) => !W.has(id))
        const live = (s.claims || []).filter((c) => !TERM.has(c.status) && !un.items.has(c.id))
        // 隔離(出處支持未能判定;檢定標準 ES9 存量)另立一節,不混入出處已撤
        const lost = (s.claims || []).filter((c) => !TERM.has(c.status) && un.items.has(c.id) && !un.held.has(c.id))
        const heldList = (s.claims || []).filter((c) => !TERM.has(c.status) && un.held.has(c.id))
        const lv = (x) => x?.evidence?.level || UNASSESSED
        const claimOf = new Map((s.claims || []).map((c) => [c.id, c]))
        const src = (ids0) => {
            const ids = alive(ids0)
            const list = ids.slice(0, 3).map((id) => `[[${id}]]`)
            return list.length ? `${list.join('、')}${ids.length > 3 ? ` 等 ${ids.length} 篇` : ''}` : '（無）'
        }
        const goneTitles = (ids) => (ids || []).filter((id) => W.has(id)).map((id) => s.withdrawn[id]?.title || id)
        const claimLine = (c) => {
            const meta = [lv(c) === UNASSESSED ? '證據未評估' : `證據${lv(c)}`, c.status === 'contested' ? '有爭議' : ''].filter(Boolean).join('｜')
            const lines = [`- 〔${c.id}〕${weakOf(c)}${c.text}${claimMark(c, notesById, noTopOf)}（${meta}）`]
            if ((c.conditions || []).length) lines.push(`  - 條件：${c.conditions.join('；')}`)
            if ((c.pros || []).length) lines.push(`  - 利：${c.pros.join('；')}`)
            if ((c.cons || []).length) lines.push(`  - 弊：${c.cons.join('；')}`)
            if (c.period) lines.push(`  - 期間：${c.period}`)
            if (c.critique) lines.push(`  - 銳評：${c.critique}`)
            // 出處排除集(⑳;ES9 判受支持而兩席皆未列為支持之出處):不列為出處、不計證據,另註篇數(可追溯於狀態檔)
            const ex = excludedSourcesOf(s, c.id)
            const exNote = ex.size ? `（另 ${ex.size} 篇經出處覆核判不支持此主張，不列為出處）` : ''
            const own = supportingSourcesOf(c, s, W)
            lines.push(un.held.has(c.id) ? `  - 出處：${src(own)}${exNote}（出處支持覆核未能判定：${s.held?.[c.id]?.why || '—'}）` : (un.items.has(c.id) ? `  - 出處已撤：${goneTitles(c.sources).join('、')}（筆記已刪除；待以新出處確認或撤回）` : `  - 出處：${src(own)}${exNote}`))
            return lines.join('\n')
        }
        const kindSection = (kind) => {
            const list = live.filter((c) => c.kind === kind)
            if (!list.length) return ''
            const facets = [...new Set(list.map((c) => c.facet || ''))]
            const out = [`## ${KIND_LABELS[kind] || kind}`, '']
            for (const f of facets) {
                if (f) out.push(`### ${f}`, '')
                out.push(list.filter((c) => (c.facet || '') === f).map(claimLine).join('\n'), '')
            }
            return out.join('\n').trim()
        }
        const cell = (v) => String(v ?? '').trim().replace(/\|/g, '／')
        const params = (s.parameters || []).filter((p) => !TERM.has(p.status))
        const CT = CORE_SECTION_TITLES
        const paramTable = params.length
            ? [`## ${CT.parameters}`, '', '| 編號 | 參數 | 值／區間 | 條件 | 期間 | 證據 | 出處 |', '| --- | --- | --- | --- | --- | --- | --- |',
                ...params.map((p) => `| ${p.id} | ${cell(p.name)} | ${cell(p.value)}${p.snapshot ? '（時點值）' : ''} | ${cell(p.conditions)} | ${cell(p.period)} | ${cell(lv(p))} | ${cell(un.items.has(p.id) ? '（出處已撤）' : src(p.sources))} |`)].join('\n')
            : ''
        const disputes = (s.disputes || []).filter((d) => d.status !== 'retracted')
        // 各方之出處＝直接出處 ∪ 所引主張之出處(只引既有主張之一方不顯示「（無）」)
        const claimSrc = new Map((s.claims || []).map((c) => [c.id, c.sources || []]))
        const sideSrc = (sd) => [...new Set([...(sd.sources || []), ...(sd.claims || []).flatMap((cid) => claimSrc.get(cid) || [])])]
        const disputeSec = disputes.length
            ? [`## ${CT.disputes}`, '', ...disputes.map((d) => [
                `- 〔${d.id}〕${d.question}（${d.status === 'resolved' ? '已解決' : '未解決'}）`,
                ...(d.sides || []).map((sd, k) => `  - 第${k + 1}方：${weakOf(sd)}${sd.position}${sideMark(sd, claimOf, notesById, noTopOf)}${(sd.conditions || []).length ? `（條件：${sd.conditions.join('；')}）` : ''}｜證據${sd.evidence?.level || UNASSESSED}${(sd.claims || []).length ? `｜依〔${sd.claims.join('〕〔')}〕` : ''}｜${un.sides.get(d.id)?.has(k) ? '出處已撤（筆記已刪除）' : `出處：${src(sideSrc(sd))}`}`),
                ...(d.note ? [`  - 註：${d.note}`] : []),
                // 終審存疑(未解決者常駐):終審認為兩方可能非對立而未移除,待提煉或整併以拆解處理或維持;
                //   檢定標準 ES14 之兩席覆核判為真對立者加註(存疑結案;判非對立者已拆解,不在此節)
                ...(d.status === 'open' && isobj(d.doubt) ? [`  - 終審存疑（v${d.doubt.version}${d.doubt.count > 1 ? `，累計 ${d.doubt.count} 次` : ''}；${d.doubt.reason}）：${d.doubt.note}${doubtCheckedOf(s, d) ? `；兩席覆核（v${s.doubtChecks[d.id].version}）：真對立——${s.doubtChecks[d.id].reason || '（未附說明）'}` : ''}`] : []),
            ].join('\n'))].join('\n')
            : ''
        const qs = (s.questions || []).filter((q) => q.status === 'open')
        const gone = [...(s.claims || []), ...(s.parameters || []), ...(s.disputes || []), ...(s.questions || [])].filter((x) => TERM.has(x.status))
        // 拆解之爭議(非對立):各方之內容在主張裡,列出主張編號供追溯
        const dissolvedLine = (x) => `- 〔${x.id}〕已拆解（非對立：${x.statusNote || ''}；各方見〔${[...new Set((x.sides || []).flatMap((sd) => sd.claims || []))].join('〕〔')}〕）：${x.question || ''}`
        const PENDING_VERB = { retract: '撤回', supersede: '取代', merge: '合併', dispute_dissolve: '拆解', add: '新增', dispute_add: '立爭議', param_add: '新增參數', question_add: '新增問題', revise: '改寫', confirm: '確認', contest: '列入爭議' }
        const history = [
            ...gone.map((x) => (x.status === 'retracted'
                ? (x.retractReason === DISSOLVED_RETRACT_REASON && isarr(x.sides) ? dissolvedLine(x) : `- 〔${x.id}〕已撤回（${x.retractReason}${x.mergedInto ? `，併入〔${x.mergedInto}〕` : ''}）：${x.text || x.name || x.question || ''}`)
                : `- 〔${x.id}〕已取代${x.supersededBy ? `（由〔${x.supersededBy}〕）` : ''}：${x.text || x.name || ''}｜理由：${x.statusNote || ''}${x.validPeriod ? `｜舊觀點成立期間：${x.validPeriod}` : ''}`)),
            // 已結案之待解問題亦留痕:文字中之〔Q〕引用在本文可找到,答案與出處不隨結案消失
            ...(s.questions || []).filter((q) => q.status === 'resolved').map((q) => `- 〔${q.id}〕已結案：${q.text}｜${q.resolution || ''}`),
            ...(s.pendingReview || []).map((e) => `- 待審：〔${e.op.id || e.op.into || e.op.ref || ''}〕擬${PENDING_VERB[e.op.op] || e.op.op}${e.op.by ? `（由〔${e.op.by}〕）` : ''}：${e.op.reason || e.reason || ''}`),
            ...(s.essenceHistory || []).map((e) => `- 先前之本質（v${e.version}）：${e.text}`),
        ]
        const used = [...citedNotesOf(s)]
        const usedLines = used.slice(0, 30).map((id) => {
            const n = notesById.get(id)
            return `- [[${id}]]${n ? ` ${n.title}（${n.sourceName}）` : ''}`
        })
        // 本質待重寫須呈現給讀者(否則讀者以為本質仍綜合現行主張):所引主張被取代／撤回(stale,持久)或出處已撤(衍生)
        const lostEss = essenceLostOf(s, un)
        const staleNote = [
            isobj(s.essence?.stale) ? `所引〔${s.essence.stale.from}〕已於 v${s.essence.stale.version} 被取代或撤回` : '',
            lostEss.length ? `所引〔${lostEss.join('〕〔')}〕之出處已撤` : '',
        ].filter(Boolean).join('；')
        const essence = s.essence?.text
            ? `${s.essence.text}${(s.essence.claims || []).length ? `（依〔${s.essence.claims.join('〕〔')}〕）` : ''}${staleNote ? `\n\n> 本質待重寫：${staleNote}。` : ''}`
            : '（尚未歸納）'
        const contested = live.filter((c) => c.status === 'contested').length
        const lostSec = lost.length
            ? [`## ${CT.lost}`, '', '以下條目之出處筆記已被刪除；在以新出處確認或撤回前，不宜引用。', '', lost.map(claimLine).join('\n')].join('\n')
            : ''
        const heldSec = heldList.length
            ? [`## ${CT.held}`, '', '以下條目經兩席覆核仍無法判定是否受所引出處支持，已移出有效條目、待使用者判讀；判讀前不宜引用。', '', heldList.map(claimLine).join('\n')].join('\n')
            : ''
        const body = [
            `# 核心知識：${s.concept}`,
            '',
            `> v${s.version}｜有效主張 ${live.length} 條（有爭議 ${contested}）${lost.length ? `｜出處已撤 ${lost.length} 條` : ''}${heldList.length ? `｜待判 ${heldList.length} 條` : ''}｜依據 ${used.length} 篇｜更新 ${s.updatedAt || '—'}`,
            '',
            `## ${CT.essence}\n\n${essence}`,
            kindSection('principle'),
            kindSection('rule'),
            paramTable,
            kindSection('pitfall'),
            disputeSec,
            kindSection('temporal'),
            ...claimKinds.filter((k) => !KIND_LABELS[k]).map(kindSection),
            lostSec,
            heldSec,
            qs.length ? [`## ${CT.questions}`, '', ...qs.map((q) => `- 〔${q.id}〕${q.text}${un.items.has(q.id) ? '（出處已撤）' : ''}`)].join('\n') : '',
            (s.related || []).length ? [`## ${CT.related}`, '', ...s.related.map((r) => `- ${r}`)].join('\n') : '',
            history.length ? [`## ${CT.history}`, '', ...history].join('\n') : '',
            used.length ? [`## ${CT.sources}`, '', ...usedLines, ...(used.length > 30 ? [`- …共 ${used.length} 篇（完整清單在核心狀態檔）`] : [])].join('\n') : '',
        ].filter(Boolean).join('\n\n')
        return { front: { title: `核心知識：${s.concept}`, related_concepts: (s.related || []).slice(0, 8) }, body }
    }

    // ── 對外之提示詞:有覆寫者交覆寫函數(opt.prompts),否則用內建;覆寫之回傳於呼叫期檢查(見 EXTRACT_PROMPT_SIGS)──

    /**
     * 可覆寫之成員(抓取套件之 overridablePrompt):未覆寫即內建函數本身;覆寫者收原參數(經與內建同一之正規化)＋末參數 ctx,回傳經檢查
     *
     * @param {String} key 輸入 opt.prompts 之鍵
     * @param {Function} builtin 輸入內建函數(即 ctx.defaultPrompt)
     * @param {Function} check 輸入回傳之檢查 (值, 鍵) => 值(不合即拋)
     * @param {Function} [norm] 輸入原參數之正規化 (...原參數) => 參數陣列,未給即原樣
     * @returns {Function} 回傳與內建同參數之函數
     */
    const overridable = (key, builtin, check, norm) => overridablePrompt(custom, key, builtin, { check, norm, ctxOf })
    const listArg = (xs) => [isarr(xs) ? xs : []]
    // 整合、審計、修訂之內建實作(domain/prompts)之稱呼:一律帶入本 domain 之 kb(與其他各步同一稱呼;內建、覆寫與覆寫所取之 defaultPrompt 皆收到同一 ctx.kb)
    const withKb = (ctx) => [{ ...(isobj(ctx) ? ctx : {}), kb }]
    const kbBuiltin = (fn) => (ctx) => fn(...withKb(ctx))

    return {
        kb, // 知識庫稱呼(vocab.kbLabel／domain):各步之提示詞同一稱呼
        claimKinds,
        limits,
        rules,
        evidence: makeEvidence({ levels, caps: vocab.evidenceCaps, basisCaps: vocab.basisCaps, selfLimit: vocab.selfLimitPhrases }),
        weakEvidence: g.weakEvidence, // 弱證據之說法(規則「…的參數須在 conditions 註明此限制」;檢定標準 ES13 亦認之為限制語)
        evidenceRule: vocab.evidenceRule, // 最高證據等級之要件(推定之判準;呼叫端據以供給 renderState 之 ctx.noTopOf)
        toneOf,
        noteDigest,
        contentKeyVersion: CONTENT_KEY_VERSION, // 內容鍵版本(noteDigest 之 contentKey;換版即重設基準)
        buildRecheckText: overridable('recheckText', buildRecheckText, maybeEmptyTextOf, (items, o) => [isarr(items) ? items : [], isobj(o) ? o : {}]), // 重審段(只在有重審筆記時出現)
        withdrawnRule: overridable('withdrawnRule', withdrawnRule, ruleOrNullOf, () => []), // 條件規則(只在有出處已撤之條目時加入)
        buildProposePrompt: overridable('propose', buildProposePrompt, promptTextOf),
        buildIntegratePrompt: overridable('integrate', kbBuiltin(buildIntegratePrompt), promptTextOf, withKb), // 稱呼取本 domain 之 kb(閉包;呼叫端不另給)
        buildAuditPrompt: overridable('audit', kbBuiltin(buildAuditPrompt), promptTextOf, withKb),
        buildRevisePrompt: overridable('revise', kbBuiltin(buildRevisePrompt), promptTextOf, withKb),
        buildReviewPrompt: overridable('review', buildReviewPrompt, promptTextOf),
        buildAcceptAppendix: overridable('acceptAppendix', buildAcceptAppendix, maybeEmptyTextOf),
        buildConsolidatePrompt: overridable('consolidate', buildConsolidatePrompt, promptTextOf),
        buildEssencePrompt: overridable('essence', buildEssencePrompt, promptTextOf), // 本質專步(檢定標準 ES6)
        buildSupportCheckPrompt: overridable('supportCheck', buildSupportCheckPrompt, promptTextOf), // 出處支持覆核(檢定標準 ES9;新進與存量)
        buildSupportRevisePrompt: overridable('supportRevise', buildSupportRevisePrompt, promptTextOf), // 依出處收斂措辭之修訂(ES9 之語氣超出證據)
        buildOrphanCheckPrompt: overridable('orphanCheck', buildOrphanCheckPrompt, promptTextOf), // 失撐主張找支持(檢定標準 ES8)
        buildDisputeCheckPrompt: overridable('disputeCheck', buildDisputeCheckPrompt, promptTextOf), // 爭議覆核(檢定標準 ES14)
        buildSynonymProposePrompt: overridable('synonymPropose', buildSynonymProposePrompt, promptTextOf, listArg), // 同義核心之提議(檢定標準 ES11)
        buildSynonymCheckPrompt: overridable('synonymCheck', buildSynonymCheckPrompt, promptTextOf, listArg), // 同義核心之覆核(檢定標準 ES11)
        renderState,
        renderTwinRedirect: twinRedirectBodyOf, // 分身轉址頁之本文(內建版型)
    }
}
